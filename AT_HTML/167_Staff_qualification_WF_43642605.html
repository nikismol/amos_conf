Рассылка настроена в рамках TQ 33029.

Результат: Отчет для руководства ДПЛГ с информацией из APN 551:
     1. список сотрудников, имеющих выбранную квалификацию - issue date, expire date
     2. список сотрудников записанных на курс с выбранной квалификацией, старт курса.

Рассылка выполняется, если у сотрудника заканчивается спецификация (Expire Date) типа CAD_TRAINING.

Разработчик: Иваненко Владимир Андреевич.

SELECT 
		sign.user_sign AS Sign,
		sign.firstname||' '||sign.lastname AS Name,
        staff_pqs_class.pqs_class_id||'/'||staff_pqs_type.type_label AS Qualification,
        to_char(to_date('01.01.1970', 'dd.mm.yyyy') + (staff_pqs_qualification.issue_date+729)) Issue_Date,
       	to_char(to_date('01.01.1970', 'dd.mm.yyyy') + (staff_pqs_qualification.expiry_date+729)) Expire_Date
FROM staff_pqs_qualification
		LEFT JOIN staff_pqs_type ON staff_pqs_qualification.pqs_type_no_i = staff_pqs_type.pqs_type_no_i
		LEFT JOIN staff_pqs_class ON staff_pqs_type.pqs_class_no_i = staff_pqs_class.pqs_class_no_i 
		LEFT JOIN sign ON staff_pqs_qualification.employee_no_i = sign.employee_no_i 
WHERE 
 staff_pqs_class.pqs_class_id = 'CAD_TRAINING'
AND
staff_pqs_qualification.expiry_date = FLOOR(sysdate-to_date('19711231','yyyymmdd'))+30

UNION 

SELECT 'EXP_CHK_OK',
		NULL,
       	NULL,
	   	NULL,
		NULL
FROM dual 
WHERE (
         SELECT count(1) 
         FROM staff_pqs_qualification
		LEFT JOIN staff_pqs_type ON staff_pqs_qualification.pqs_type_no_i = staff_pqs_type.pqs_type_no_i
		LEFT JOIN staff_pqs_class ON staff_pqs_type.pqs_class_no_i = staff_pqs_class.pqs_class_no_i 
		LEFT JOIN sign ON staff_pqs_qualification.employee_no_i = sign.employee_no_i 
WHERE 
 staff_pqs_class.pqs_class_id = 'CAD_TRAINING'
AND
staff_pqs_qualification.expiry_date = FLOOR(sysdate-to_date('19711231','yyyymmdd'))+30
      )= '0' 
ORDER BY 4
