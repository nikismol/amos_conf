AMOS Team : Порядок обезличивания персональных данных

Created by Филипенков Никита Константинович, last modified on авг. 15, 2023

 Персональные данные обезличиваются в системе администраторами AMOS на основании запросов ключевых пользователей или координаторов в APN 869. Запросы на обезличивание персональных данных должны содержать имена учетных записей AMOS (sign.user_sign). Для обезличивания персональные данные заменяются на «F_xxxxxxx», где x – случайная буква латинского алфавита. Для обезличивания персональных данных осуществляется замена данных в следующих полях в таблице “SIGN” базы данных AMOS поля:

  • employee_no
  • firstname
  • lastname
  • birthdate
  • email
  • badge_number
  • notes
  • contract_id
  • user_sign

В таблице «rm_property_value» базы данных AMOS поле char_value


SQL для обезличивания данных
/* 
** Проверить на стади сначала, потом на выполнить на проде 
*/
alter session set current_schema = amosserver_study;


/*
** 1 - готовлю DML-запросы для обновления таблицы sign заданных логинов
*/
SELECT 'UPDATE sign
           SET firstname ='|| CHR(39) || 'F_' || CHR(39) ||  ' ||DBMS_RANDOM.STRING ('|| CHR(39) || 'l' || CHR(39) ||', 7),
               lastname =' || CHR(39) || 'F_' || CHR(39) || ' ||DBMS_RANDOM.STRING  ('|| CHR(39) || 'l' || CHR(39) ||', 7),
               description = NULL,
               birthdate = NULL,
               email = NULL,
               badge_number = NULL,
               notes = null,
               url_to_pic = null
               where user_sign = ' || CHR(39) || sign.user_sign || CHR(39)||';' as sss
FROM sign
WHERE 1=1
      AND sign.user_sign IN ();
      
/*
** 2 - Выполнить update-запросы сформированные в п.1 и !!!ВЫПОЛНИТЬ COMMIT!!! 
*/      
  

/*
** 3 - изменяю ФИО в rm_property_value (указать список логинов) 
*/              
SELECT 'UPDATE rm_property_value
           SET char_value = CASE WHEN property_type_no_i = 42 THEN '|| CHR(39) ||sign.lastname|| CHR(39) ||
                                 ' WHEN property_type_no_i = 41 THEN' || CHR(39) || 'F_' || CHR(39) ||  ' ||DBMS_RANDOM.STRING ('|| CHR(39) || 'l' || CHR(39) ||', 7)
                                 ELSE '|| CHR(39) ||sign.firstname|| CHR(39) ||' END 
         WHERE property_value_no_i = ' || rm_property_value.property_value_no_i ||';'as sss
FROM sign
JOIN rm_property_value ON sign.employee_no_i = rm_property_value.owner_amos_key AND rm_property_value.owner_amos_type = 'SG' AND property_type_no_i IN (40, 41, 42)
WHERE 1=1
      AND sign.user_sign IN ();
