INSERT INTO AFL_ATECH_MEL_MONITOR
SELECT DISTINCT
    sysdate as current_sysdate
    , FLOOR(sysdate-to_date('19711231','yyyymmdd')) as current_amosdate
    , woh.event_perfno_i as wo_event_perfno_i
    , woh.ac_registr as wo_ac_registr
    , a.ac_typ as wo_ac_typ
    , a.ac_model as wo_ac_model
    , woh.issue_sign as wo_issue_sign
    , wot.user_sign as wo_transfer_sign
    , wot.reason_code as wo_transfer_reason_code
    , wot.event_transferno_i as wo_current_event_transferno_i
    , DECODE(woh.type, 'M', 'Maint', 'P', 'PiRep', 'C', 'Cabin', 'Unknown') as wo_type
    , wph.wpno_i as wo_current_wpno_i
    , pr.part_requestno_i as wo_part_requestno_i
    , pr.partno as part_request_partno
    , pr.status as part_request_status
    , pf.status as part_forecast_status
FROM 
    wo_header woh
LEFT JOIN
    wo_transfer wot ON woh.event_perfno_i = wot.event_perfno_i  AND wot.is_last_transfer = 'Y'
LEFT JOIN
    aircraft a ON a.ac_registr = woh.ac_registr
LEFT JOIN
    wp_assignment wps ON wps.event_perfno_i = woh.event_perfno_i
LEFT JOIN
    wp_header wph ON wps.wpno_i = wph.wpno_i
LEFT JOIN
    workstep_link wsl ON wsl.event_perfno_i = woh.event_perfno_i
LEFT JOIN 
    part_request_2 pr ON pr.status = 0 AND pr.event_type = 'WSL' AND pr.event_key = wsl.workstep_linkno_i
LEFT JOIN
    part_forecast pf ON pr.part_requestno_i = pf.part_requestno_i
WHERE 
    woh.state = 'O'
    AND woh.mel_code IS NOT NULL
    AND woh.type IN ('M', 'P', 'C')
