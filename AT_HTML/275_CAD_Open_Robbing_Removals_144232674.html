AMOS Team : 275_CAD_Open_Robbing_Removals

Created by Филипенков Никита Константинович on сент. 26, 2023 


SELECT aircraft.ac_typ as ac_typ,
              h_ya.ac_registr as ac_registr,
              CASE WHEN REGEXP_LIKE(h_ya.partno, '^[0-9]+$') THEN chr(39)||h_ya.partno ELSE h_ya.partno END as partno,
              part.description as description,
              CASE WHEN REGEXP_LIKE(h_ya.serialno, '^[0-9]+$') THEN chr(39)||h_ya.serialno ELSE h_ya.serialno END as serialno,
              TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + h_ya.del_date, 'dd.mm.yyyy') as removal_date,
              h_ya.event_perfno_i as removal_wo_no,
              wo_header.cannibalized as is_cannibalized_wo,
              DECODE(wo_header.state, 'O', 'Open', 'C', 'Closed', wo_header.state) as wo_status,
              TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + ac_util.departure_date, 'dd.mm.yyyy') as last_flight_date,
              aircraft.last_station as last_station,
              wp.wpno as current_wp 
FROM (
             SELECT history.historyno_i,
                           MAX(history.historyno_i) OVER (PARTITION BY history.ac_registr, history.location) as max_historyno_i,
                           history.ac_registr,
                           history.location,
                           history.labelno,
                           history.partno,
                           history.serialno,
                           history.del_date,
                           history.event_perfno_i
             FROM history
             WHERE history.vm = 'YA'
                           AND history.del_date >= 17533 /* period from: 01.01.2020 */
                           AND history.label_start = 'Y'
                           /* AND (REGEXP_LIKE(history.ac_registr, REPLACE('@VAR.v02_ac_registr@', ', ', '|')) OR '@VAR.v02_ac_registr@' IS NULL) */
            ) h_ya
LEFT JOIN part ON h_ya.partno = part.partno
LEFT JOIN history h_ye ON h_ya.labelno = h_ye.labelno AND h_ye.vm = 'YE'
LEFT JOIN wo_header ON h_ya.event_perfno_i = wo_header.event_perfno_i AND wo_header.state <> 'V'
LEFT JOIN rotables ON h_ya.location = rotables.location AND rotables.ac_registr = h_ya.ac_registr
JOIN aircraft ON h_ya.ac_registr = aircraft.ac_registr AND aircraft.status <> 9 AND aircraft.operator = 'AFL'
LEFT JOIN (
                   SELECT ac_utilization.ac_registr,
                                 ac_utilization.departure_date,
                                 MAX(ac_utilization.departure_date) OVER (PARTITION BY ac_utilization.ac_registr) as max_departure_date,
                                 ac_utilization.recno,
                                 MAX(ac_utilization.recno) OVER (PARTITION BY ac_utilization.ac_registr, ac_utilization.departure_date) as max_recno
                   FROM ac_utilization
                   WHERE 1=1
                                  AND ac_utilization.recno > 0
                                  AND ac_utilization.null_leg_commited <> 1
                   ) ac_util ON h_ya.ac_registr = ac_util.ac_registr AND ac_util.departure_date = ac_util.max_departure_date AND ac_util.recno = ac_util.max_recno
LEFT JOIN (
                   SELECT wp_header.ac_registr,
                                 LISTAGG(wp_header.wpno ||' (' || wp_header.description || ')', '; '||CHR(10)) WITHIN GROUP (ORDER BY wp_header.wpno) as wpno
                   FROM wp_header
                   WHERE 1=1
                                  AND wp_status IN (-1, -3) 
		     AND (/* выбираю пакеты, которые "выполняются" на момент выгрузки отчета */ 
                                            NVL(case
                                                     when wp_header.act_start_date>0 and wp_header.act_start_time>0
                                                     then TO_TIMESTAMP_TZ('31.12.1971 00:00:00 00:00', 'DD.MM.YYYY HH24:MI:SS TZH:TZM') + numtodsinterval(wp_header.act_start_date, 'day') + NUMTODSINTERVAL(wp_header.act_start_time * 60, 'SECOND')
                                                     else null
                                                     end,
                                                     TO_TIMESTAMP_TZ('31.12.1971 00:00:00 00:00', 'DD.MM.YYYY HH24:MI:SS TZH:TZM') + numtodsinterval(wp_header.start_date, 'day') + NUMTODSINTERVAL(wp_header.start_time * 60, 'SECOND')
                                                    ) AT TIME ZONE 'Europe/Moscow'
                                                    BETWEEN from_tz(CAST (TRUNC(systimestamp)+ numtodsinterval(0, 'day') AS TIMESTAMP), 'Europe/Moscow')
                                                                       AND from_tz(CAST (TRUNC(systimestamp)+ numtodsinterval(0, 'day') AS TIMESTAMP), 'Europe/Moscow')
                                               OR
                                            NVL(
                                                     case
                                                     when wp_header.act_start_date>0 and wp_header.act_start_time>0
                                                     then TO_TIMESTAMP_TZ('31.12.1971 00:00:00 00:00', 'DD.MM.YYYY HH24:MI:SS TZH:TZM') + numtodsinterval(wp_header.act_start_date, 'day') + NUMTODSINTERVAL(wp_header.act_start_time * 60, 'SECOND')
                                                     else null
                                                     end,
                                                     TO_TIMESTAMP_TZ('31.12.1971 00:00:00 00:00', 'DD.MM.YYYY HH24:MI:SS TZH:TZM') + numtodsinterval(wp_header.start_date, 'day') + NUMTODSINTERVAL(wp_header.start_time * 60, 'SECOND')
                                                    ) AT TIME ZONE 'Europe/Moscow' < from_tz(CAST (TRUNC(systimestamp)+ numtodsinterval(0, 'day') AS TIMESTAMP), 'Europe/Moscow')
                                            AND
                                            NVL(case 
                                                     when wp_header.act_end_date>0 and wp_header.act_end_time>0
                                                     then TO_TIMESTAMP_TZ('31.12.1971 00:00:00 00:00', 'DD.MM.YYYY HH24:MI:SS TZH:TZM') + numtodsinterval(wp_header.act_end_date, 'day') + NUMTODSINTERVAL(wp_header.act_end_time * 60, 'SECOND')
                                                     else null
                                                     end,
                                                     TO_TIMESTAMP_TZ('31.12.1971 00:00:00 00:00', 'DD.MM.YYYY HH24:MI:SS TZH:TZM') + numtodsinterval(wp_header.end_date, 'day') + NUMTODSINTERVAL(wp_header.end_time * 60, 'SECOND')
                                                    ) AT TIME ZONE 'Europe/Moscow' >= from_tz(CAST (TRUNC(systimestamp)+ numtodsinterval(0, 'day') AS TIMESTAMP), 'Europe/Moscow')
                                          )
                   GROUP BY wp_header.ac_registr
                   ) wp ON h_ya.ac_registr = wp.ac_registr
WHERE 1=1
              AND h_ya.historyno_i = h_ya.max_historyno_i
              AND h_ye.historyno_i IS NULL
              AND rotables.psn IS NULL
              AND h_ya.del_date >= ac_util.departure_date
              AND wo_header.type = 'S'
ORDER BY 1, 2, 3, 5
