  • Ниже представлен SQL запрос для указания в Selector Options:

    SELECT 
    ord.orderno_i, 
    ord.orderno
FROM 
    (/* Specified orders in state "open" or draft orders (T) */
        SELECT 
            od_header.orderno_i, 
            od_header.orderno,         
            to_date('31.12.1971', 'dd.mm.yyyy') + NVL(GREATEST(
                (/* get last od_detail change */
                SELECT MAX(od_detail.mutation + od_detail.mutation_time / 24 / 60 / 60 / 1000)
                FROM od_detail 
                WHERE 1=1
                  AND od_detail.orderno_i = od_header.orderno_i
                  AND od_detail.mutator NOT IN ('MIGRUSR', 'RAMAX_AC', 'AMOS_SRV')
                ),
                NVL(
                    (/* get last "issued by address" change - the only field in od_header that needs to be checked*/
                        SELECT 
                            MAX(bohi_changes.changes_date + bohi_changes.changes_time / 24 / 60) as address_issue_last_change_date
                        FROM bohi_changes
                        WHERE 1=1
                          AND bohi_changes.action = 'Saved from OrderSaveAction'
                          AND bohi_changes.user_sign NOT IN ('MIGRUSR', 'RAMAX_AC', 'AMOS_SRV')
                          AND bohi_changes.changes LIKE ('%<transactionType>UPDATE</transactionType>%')
                          AND bohi_changes.changes LIKE ('%<name>address_issue</name>%')                
                          AND bohi_changes.versionno_i in (
                            SELECT bohi_version.versionno_i
                            FROM bohi_version
                            WHERE 1=1
                              AND bohi_version.primkey = od_header.orderno_i
                              AND bohi_version.mime_type = 'ORDER'  
                            )  
                        ), 
                    0)
                ),
                od_header.mutation + od_header.mutation_time / 24 / 60 / 60 / 1000) as last_mutation_date, /* Last mutation date and time */
            od_header.ext_state,
            od_header.order_type,
            od_header.vendor
        FROM od_header 
        WHERE 1=1
          AND (to_date('31.12.1971','DD.MM.YYYY') + od_header.mutation) > trunc(sysdate-7)
          AND od_header.mutator NOT IN ('MIGRUSR', 'RAMAX_AC', 'AMOS_SRV')
          AND od_header.vendor NOT IN ('CLOSED', 'TECH_V', 'DUZD_TECH', 'DUZD_WSP') 
          AND (
              od_header.order_type IN ('P', 'PX', 'PR', 'PU', 'L', 'LU', 'Y', 'YU', 'R', 'X', 'RC', 'RU', 'CP', 'CU') 
                  AND (
                      od_header.state = 'O' AND od_header.ext_state in ('O', 'BO')
                          OR
                      od_header.state = 'C' AND od_header.ext_state = 'B'
                      )
                  OR  od_header.order_type = 'T' 
                          AND od_header.state = 'O' 
                          AND od_header.ext_state in ('B', 'BO')
                  OR od_header.order_type = 'D'
              )
        ) ord
        LEFT JOIN (
                    /* Transfer history */
            SELECT hist.source_key,
                   MAX(to_date('31.12.1971', 'dd.mm.yyyy') + hist.mutation + hist.mutation_time / 24 / 60 / 60 / 1000) last_mutation_date_time  /* Last date of transfer for source_key */
            FROM aim_transfer_history hist
            WHERE 1=1
              AND hist.success = 'Y'
              AND hist.source_type = 'ORDER'
              AND hist.service = 'TRANSFER_ORDER'
              AND hist.source_key is not null
              AND (hist.aim_configno_i = '6')
            GROUP BY hist.source_key
            ) hist ON ord.orderno = hist.source_key
WHERE 1=1
  AND ord.last_mutation_date > TRUNC(sysdate - 7)
  AND ( 
      ord.last_mutation_date > hist.last_mutation_date_time AND ord.ext_state = 'O' /* Changed orders (been already loaded) */
      OR
      hist.source_key is null AND ord.ext_state in ('B', 'O', 'BO') /* New Orders (never been loaded) */
      ) 


;

Настройка селектора в APN 1404

  • В Configuration Detail выбираем подвкладку selector и в окне Export Selector заполнить данные:

  • Ниже представлен SQL запрос для указания в Selector Options:

    select 
    ord.orderno
from /* Specified orders in state "open" or draft orders (T) */
       (
       select 
           hr.orderno_i,
           mutator,
           hr.orderno,
           internal_repair,
           /* Last mutation date and time */
           to_date('31.12.1971', 'dd.mm.yyyy') + nvl((select max(dl.mutation + dl.mutation_time / 24 / 60 / 60 / 1000) from od_detail dl where dl.orderno_i = hr.orderno_i), hr.mutation + hr.mutation_time / 24 / 60 / 60 / 1000) last_mutation_date,
           hr.ext_state,
           hr.order_type,
           hr.vendor
       from od_header hr
       where hr.order_type in ('P', 'PX', 'PR', 'PU', 'L', 'LU', 'Y', 'YU', 'R', 'X')) ord
           left join /* Transfer history */(
           select 
               hist.source_key,
               /* Last date of transfer for source_key */
               max(to_date('31.12.1971', 'dd.mm.yyyy') + hist.mutation +
                   hist.mutation_time / 24 / 60 / 60 / 1000) last_mutation_date_time
           from aim_transfer_history hist
           where hist.success = 'Y'
             and hist.source_type = 'ORDER'
             and hist.service = 'TRANSFER_ORDER'
             and hist.source_key is not null
             and hist.aim_configno_i = '809'
           group by hist.source_key
           ) hist on ord.orderno = hist.source_key
where /* Changed orders (been already loaded)*/
    (
        ord.last_mutation_date > hist.last_mutation_date_time 
            OR  
        hist.source_key is null
        ) 
  and ord.last_mutation_date > to_date('01.09.2020', 'dd.mm.yyyy') 
  AND ord.internal_repair = 'N'
  AND ord.mutator <> 'MIGRUSR' 
  and ord.mutator <> 'RAMAX_AC' 
  and ord.mutator <> 'AMOS_SRV' 
  and ord.mutator <> 'ADMINRMX' 
  AND ord.vendor != 'TECH_V' 
  AND ord.vendor <> 'CLOSED' 
  AND ord.vendor != 'OPEN' 
  AND ord.ext_state <> 'PR' 
  AND ord.ext_state <> 'R'
  AND ord.vendor IS NOT NULL 
  and ascii(ord.vendor) <> 49824
  AND ord.vendor <> 'CMD_ATEC' 
  AND ord.vendor <> 'CMD_BRS' 
  AND ord.vendor <> 'CMD_BS' 
  AND ord.vendor <> 'C&NES' 
  AND ord.vendor <> 'CMD_EES'
  AND ord.vendor <> 'CMD_EPES' 
  AND ord.vendor <> 'CMD_FS' 
  AND ord.vendor <> 'GES' 
  AND ord.vendor <> 'CMD_OEES' 
  AND ord.vendor <> 'CMD_WS'
  AND ord.vendor <> 'FIPS' 
  AND ord.vendor <> 'SRS' 
  AND ord.vendor <> 'SRT-S7_CH317'
