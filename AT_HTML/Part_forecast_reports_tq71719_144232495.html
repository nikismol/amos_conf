SELECT
      DECODE (part_forecast.display_event_type,
                        'GENEVENT', 'Generic Event',
                        'DOP', 'Pending Document',
                        'CH', 'Check Pending',
                        'WO', 'Workorder',
                        'SRT', 'Single Running Taskcard',
                        'CM_WOSC', 'CM Workstep',
                        'EFL', 'Effectivity Link',
                        'MER', 'Replacement Event',
                        'TC', 'Taskcard',
                        'ORDER_DE', 'Order Detail',
                        part_forecast.display_event_type) as event_type
      ,part_forecast.event_display as event
      ,part_forecast.receiver as receiver
      ,CASE WHEN REGEXP_LIKE(part_forecast.partno, '^[0-9]+$') THEN chr(39)||part_forecast.partno ELSE part_forecast.partno END as partno
      ,part.mat_class
      ,part_forecast.part_description as part_description
      ,part_forecast.part_special_types as special_types
      ,DECODE (part_forecast.status,
                         0, 'Not calculated',
                         1, 'Unavailable', 
                         2, 'Ordered',
                         3, 'Ordered and Linked', 
                         4, 'Needs Transfer', 
                         5, 'In Transfer', 
                         6, 'Available',
                         7, 'On Station', 
                         8, 'Externally Provisioned',
                         9, 'Part Request Fulfilled',
                         10, 'Calculation Error',
                         20, 'Available Partly Linked',
                         30, 'Available Linked'
      )
      ,CASE WHEN SUBSTR(special_flags, 3,1)='Y' THEN 'Y' ELSE '' END as is_DD_wo
      ,CASE WHEN SUBSTR(special_flags, 4,1)='Y' THEN 'Y' ELSE '' END as is_MEL_wo
      ,CASE WHEN SUBSTR(special_flags, 10,1)='Y' THEN 'Y' ELSE '' END as is_cannibalized_wo
      ,CASE WHEN SUBSTR(special_flags, 11,1)='Y' THEN 'Y' ELSE '' END as is_AOG_wo
      ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + part_forecast.expected_date, 'DD.MM.YYYY') as due_date
      ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + part_forecast.planned_date, 'DD.MM.YYYY') as planned_date
      ,TO_NUMBER(part_forecast.requested_qty) as requested_qty
      ,part_forecast.measure_unit as measure_unit
      ,TO_NUMBER(part_forecast.total_requested_qty) as total_requested_qty
      ,TO_NUMBER(part_forecast.accessable_system_qty) as accessable_system_qty
      ,TO_NUMBER(part_forecast.accessable_station_qty) as accessable_station_qty
      ,TO_NUMBER(part_forecast.total_reserved_order_qty) as total_reserved_order_qty
      ,TO_NUMBER(part_forecast.total_confirmed_order_qty) as total_confirmed_order_qty
      ,TO_NUMBER(part_forecast.total_order_qty) as total_order_qty
      ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + part_forecast.best_order_date, 'DD.MM.YYYY') as delivery_date
      ,part_forecast.wpno as workpackage
FROM part_forecast
LEFT JOIN aircraft ON part_forecast.receiver=aircraft.ac_registr
LEFT JOIN part on part_forecast.partno = part.partno
WHERE (part_forecast.part_request_part_cond <> 'US' OR part_forecast.part_request_part_cond IS NULL)
    AND part_forecast.receiver_type = 'AC'
    AND (
            part_forecast.expected_date <= FLOOR(TRUNC(ADD_MONTHS(sysdate, 6), 'dd')-to_date('19711231','yyyymmdd'))
            OR part_forecast.planned_date <= FLOOR(TRUNC(ADD_MONTHS(sysdate, 6), 'dd')-to_date('19711231','yyyymmdd'))
        )
