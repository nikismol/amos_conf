SELECT
    ac_registr,
    FN_carrier,
    FN_NUMBER,
    TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + sched_departure_date, 'dd.mm.yyyy') sched_departure_date,
    TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + estim_departure_date, 'dd.mm.yyyy') estimate_departure_date,
    TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + sched_arrival_date, 'dd.mm.yyyy') sched_arrival_date,
    TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + estim_arrival_date, 'dd.mm.yyyy') estimate_arrival_date
FROM ac_future_flights
WHERE
    ac_future_flights.sched_departure_date <= TRUNC(SYSDATE -1 - TO_DATE('31.12.1971', 'dd.mm.yyyy'))
  AND ac_future_flights.sched_departure_date >= 18653
  AND FLOOR(SYSDATE-TO_DATE('19711231','yyyymmdd'))*(24*60) + TO_CHAR(SYSTIMESTAMP, 'SSSSS')/60 >= NVL(ac_future_flights.estim_departure_date, ac_future_flights.sched_departure_date)*24*60 + NVL(ac_future_flights.estim_arrival_time, ac_future_flights.sched_arrival_time) + 180
UNION
SELECT
    'EXPORT_CHECK',
    Null,
    Null,
    Null,
    Null,
    Null,
    Null
FROM dual
WHERE (
    SELECT
        count(1)
    FROM ac_future_flights
    WHERE
        ac_future_flights.sched_departure_date <= TRUNC(SYSDATE -1 - TO_DATE('31.12.1971', 'dd.mm.yyyy'))
      AND ac_future_flights.sched_departure_date >= 18653
      AND FLOOR(SYSDATE-TO_DATE('19711231','yyyymmdd'))*(24*60) + TO_CHAR(SYSTIMESTAMP, 'SSSSS')/60 >= NVL(ac_future_flights.estim_departure_date, ac_future_flights.sched_departure_date)*24*60 + NVL(ac_future_flights.estim_arrival_time, ac_future_flights.sched_arrival_time) + 180 ) = '0'
