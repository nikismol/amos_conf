SELECT 
    'V',
    moc_daily_records.dailyrecno_i as Event,
    moc_daily_records.header as header,
    moc_daily_records.description,
    DECODE(moc_daily_records.closed, 'Y', 'Status Close', 'N', 'Status Open') as status,
    moc_daily_records.ac_registr,
    moc_daily_records.station,
    moc_daily_records.flightno,
    moc_daily_records.ata_chapter,
    moc_daily_records.created_date,
    moc_daily_records.createdtime,
    moc_daily_records.closing_date,
    moc_daily_records.closing_time,
    to_char(to_date('01.01.1970', 'dd.mm.yyyy') + (moc_daily_records.created_date+729), 'dd.mm.yyyy') as creat_date
FROM moc_daily_records
WHERE moc_daily_records.event_group = 'FDM'
  AND moc_daily_records.closed = 'N'
