    SQL запрос(NEW)
    SELECT distinct history.historyno_i
FROM history
LEFT JOIN od_header ON history.orderno = od_header.orderno
LEFT JOIN od_rec_header ON DECODE(
                                                                     LENGTH(TRIM(TRANSLATE(SUBSTR(history.voucherno, 3),`1234567890`,` `))),
                                                                     null, substr(history.voucherno, 3),
                                                                     `0`
                                                                    ) = od_rec_header.receivingno   /*могут быть не только числа*/
LEFT JOIN aim_transfer_history ON history.historyno_i = aim_transfer_history.source_key     /*Transfer history*/
                                                               AND aim_transfer_history.success = `Y`
                                                               AND aim_transfer_history.source_type = `HI`
                                                               AND aim_transfer_history.service = `EXPORT_COMPONENT_HISTORY`
                                                               AND (aim_transfer_history.aim_configno_i = `2`)
                                                               AND aim_transfer_history.source_key IS NOT NULL
LEFT JOIN history_text ON history.historyno_i = history_text.historyno_i AND short_text like `Previous Location: %`
 where 1=1
              AND to_date(`31.12.1971`, `dd.mm.yyyy`) + history.created_date >= TRUNC(sysdate - 30)
              AND to_date(`31.12.1971`, `dd.mm.yyyy`) + history.del_date > to_date(`01.03.2022`, `dd.mm.yyyy`)
              AND (history.store != `EXTERNAL` OR history.store IS NULL)
              AND aim_transfer_history.source_key IS NULL
              AND history.created_by <> `MIGRUSR`
              AND (
                        history.vm in (`BN`,`CE`,`CR`,`CS`,`D`,`EA`,`ES`,`LC`,`LE`,`PS`,`SC`,`SX`,`TC`,`TS`,`TR`,`B8`,`YA`,`SI`,`SR`,`RS`,`SE`,`YR`,`DE`, `SN`, `T`)
                        OR
                        (
                         history.vm = `B1`
                         AND od_rec_header.awbno IN (`RELEASE REPAIRABLE`,`RELEASE ROTABLE`)
                         AND nvl(history.higher_partno, chr(160)) in (chr(49824), chr(160))
                        )
                        OR
                        ( /* B7 (комплектование) выгружать только с пустым orderno или по R заказу */
                         history.vm = `B7`
                         AND (nvl(history.orderno, chr(160)) in (chr(49824), chr(160)) or substr(history.orderno, 1, 1) = `R`)
                        ) 
                        OR
                        (/* для внутренних ремонтов МО выгружать все же надо */
                         /* MO только без ремонтного заказа, если нет ссылки на заказ - невидимый пробел, с самолетов не делаем */
                         history.vm = `MO`
                         AND (nvl(history.orderno, chr(160)) in (chr(49824), chr(160)) OR od_header.internal_repair=`Y` )
                         AND nvl(history.station, chr(160)) not in (chr(49824), chr(160))
                        ) 
                        OR
                        ( /* иначе при загрузке данных по самолету выгрузится куча монтажей */
                         history.vm = `YE`
                         AND nvl(history.station, chr(160)) not in (chr(49824), chr(160))
                         AND nvl(history.store, chr(160)) not in (chr(49824), chr(160))
                        ) 
                        OR
                        ( /* исключение перемещений внутри LISTOFTL */
                         history.vm IN (`T`, `LC`, `TS`, `TR`)
                         AND history.store != `LISTOFTL`
                         AND regexp_replace(history_text.short_text,`Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)`,`\2`) = `LISTOFTL` 
                        ) 
                        OR
                        (
                         history.vm IN (`AA`, `AS`)
                         AND history.store <> `123A`
                        )
                        OR
                        (
                         history.vm IN (`CI`)
                         AND od_header.order_type = `R` AND od_header.vendor = `CLOSED`
                        )
                        OR
                        (
                         history.vm in (`V`)
                         AND history.created_date >= 18899
                        )
                       )
    SQL запрос(OLD)
    select distinct hs.historyno_i
from history hs
left join (
    select
        orderno,
        internal_repair
    from od_header) ord on hs.orderno = ord.orderno
left join od_rec_header rec
    on decode( /* могут быть не только числа */
        length(trim(translate(substr(hs.voucherno,3),
        '1234567890',
            ' '))),
        null,
        substr(hs.voucherno, 3),
        '0') = rec.receivingno
left join /* Transfer history */ (
    select hist.source_key
    from aim_transfer_history hist
    where hist.success = 'Y'
      and hist.source_type = 'HI'
      and hist.service = 'EXPORT_COMPONENT_HISTORY'
      and hist.aim_configno_i = '2'
      and hist.source_key is not null
    group by hist.source_key) hist on hs.historyno_i = hist.source_key
left join /* from-to locations */(
    select
        historyno_i,
           /*      regexp_replace(t.short_text,
         `Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)`,
         `\1`) station_from, */
        regexp_replace(t.short_text,
                        'Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)',
                        '\2') store_from
                        /*      regexp_replace(t.short_text,`Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)`,`\3`) location_from */
    from history_text t
    where short_text like 'Previous Location: %') loc on hs.historyno_i = loc.historyno_i
where	(hs.created_date >= 17839) AND /*after 02-11-2020*/
	 (hs.vm in (
                  'BN',
                  'CE',
                  'CR',
                  'CS',
                  'D',
                  'EA',
                  'ES',
                  'LC',
                  'LE',
                  'PS',
                  'SC',
                  'SX',
                  'TC',
                  'TS',
                  'TR',
				  'B8',
                  'YA',
                  'SI',
                  'SR',
                  'RS',
                  'SE',
                  'YR',
                  'DE', 
                  'SN', 
	               'PE') 
	      or
	  (hs.vm = 'B1' and ( rec.awbno = 'RELEASE REPAIRABLE' OR rec.awbno = 'RELEASE ROTABLE' ) 
            and
	   nvl(hs.higher_partno, chr(160)) in (chr(49824), chr(160))) /* higher part - пустое aka равно пробелу или пустоте*/or (hs.vm = 'B7' and
       (nvl(hs.orderno, chr(160)) in (chr(49824), chr(160)) or
       substr(hs.orderno, 1, 1) = 'R')) /* B7 (комплектование) выгружать только с пустым orderno или по R заказу */
       or (hs.vm = 'MO' and
       (  (nvl(hs.orderno, chr(160)) in (chr(49824), chr(160))) OR ord.internal_repair='Y' ) and  /*заказ пустой И он не внутренний*/
       nvl(hs.station, chr(160)) not in (chr(49824), chr(160))) /* MO только без ремонтного заказа, если нет ссылки на заказ - невидимый пробел, с самолетов не делаем */
   /* для внутренних ремонтов МО выгружать все же надо */
       or (hs.vm = 'YE' and
       nvl(hs.station, chr(160)) not in (chr(49824), chr(160)) and
       nvl(hs.store, chr(160)) not in (chr(49824), chr(160))) /* иначе при загрузке данных по самолету выгрузится куча монтажей */
       or (hs.vm IN ('T', 'LC', 'TS', 'TR') and
       NOT (hs.store = 'LISTOFTL' and loc.store_from = 'LISTOFTL')) /* исключение перемещений внутри LISTOFTL */
       or (hs.vm IN ('AA', 'AS') AND hs.store <> '123A')
)
AND (hs.store != 'EXTERNAL' OR hs.store IS NULL)
   and hist.source_key is null
   and to_date('31.12.1971', 'dd.mm.yyyy') + hs.del_date >
       to_date('01.01.2020', 'dd.mm.yyyy')
  and hs.created_by <> 'MIGRUSR'

Настройка селектора в APN 1404

  • В Configuration Detail выбираем подвкладку selector и в окне Export Selector заполнить данные:

  • Ниже представлен SQL запрос для указания в Selector Options:

    select 
    distinct hs.historyno_i, 
             hs.del_date, 
             hs.vm, 
             hs.voucherno 
from history hs
    left join /* Transfer history */
        (
        select 
            hist.source_key
        from aim_transfer_history hist
        where hist.success = 'Y'
          and hist.source_type = 'HI'
          and hist.service = 'EXPORT_COMPONENT_HISTORY'
          and hist.aim_configno_i = '808'
          and hist.source_key is not null
        group by hist.source_key) hist on hs.historyno_i = hist.source_key
    left join /* from-to locations */
        (
        select 
            historyno_i,
            /*regexp_replace(t.short_text,'Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)','\1') station_from, */
            regexp_replace(t.short_text,'Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)','\2') store_from
            /*regexp_replace(t.short_text,'Previous Location: ([[:alnum:]]+)/([[:alnum:]]+)/(.*)','\3') location_from */
        from history_text t
        where short_text like 'Previous Location: %') loc on hs.historyno_i = loc.historyno_i
where 
    (
        hs.vm in ( 'TC',  'YA', 'CR') 
            or 
        (
            hs.vm = 'YE' 
                and
            nvl(hs.station, chr(160)) not in (chr(49824), chr(160)) 
                and
            nvl(hs.store, chr(160)) not in (chr(49824), chr(160))
            ) /* иначе при загрузке данных по самолету выгрузится куча монтажей */
            or 
        (
            (
                hs.vm IN ('T', 'LC',  'TR', 'TS')
                    or 
                hs.vm IN ('LE', 'PS', 'SX')
                    and 
                to_date('31.12.1971', 'dd.mm.yyyy') + hs.del_date >= to_date('27.09.2021', 'dd.mm.yyyy')
            )
                and 
            NOT (hs.store = 'LISTOFTL' and loc.store_from = 'LISTOFTL') /* исключение перемещений внутри LISTOFTL */
            ) 
            or 
        (
            hs.vm IN ('AA', 'AS') 
                AND hs.store <> '123A'
            )
        )
  AND (hs.store != 'EXTERNAL' OR hs.store IS NULL)
  and hist.source_key is null 
  and to_date('31.12.1971', 'dd.mm.yyyy') + hs.del_date > to_date('01.09.2020', 'dd.mm.yyyy')
  and hs.created_by <> 'MIGRUSR'
