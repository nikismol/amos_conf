Настройка селектора в APN 1404

  • В Configuration Detail выбираем подвкладку selector и в окне Export Selector заполнить данные:

  • Ниже представлен SQL запрос для указания в Selector Options:

    SELECT DISTINCT
    wp_header.wpno_i, wp_header.wpno
FROM wp_header
    LEFT JOIN
    (
        SELECT
            DISTINCT 'Y' as has_no_close_date,
                     wpa.wpno_i
        FROM wo_header woh
            LEFT JOIN wp_assignment wpa ON woh.event_perfno_i = wpa.event_perfno_i
        WHERE woh.state = 'C' AND woh.closing_date IS NULL
        ) wpcnts ON wpcnts.wpno_i = wp_header.wpno_i
    LEFT JOIN aim_transfer_history ON aim_transfer_history.aim_configno_i = 1108
            AND aim_transfer_history.source_type = 'WP'
            AND aim_transfer_history.service = 'TRANSFER_MRO_WORKPACKAGE'
            AND aim_transfer_history.source_type = 'WP'
            AND wp_header.wpno_i = aim_transfer_history.source_key_i
            AND (wp_header.mutation + (wp_header.mutation_time)/24/60/60/1000) < (aim_transfer_history.trf_date + aim_transfer_history.trf_time/24/60)
            AND aim_transfer_history.success = 'Y'
WHERE
    wp_header.created_date >= 17954
    AND wpcnts.has_no_close_date IS NULL
    AND aim_transfer_history.histno_i IS NULL
    AND rownum<=15
    AND wp_header.wpno_i not in (
                                 273570, 272248, 272856, 272067,
                                 276788, 274996, 275410, 276783,
                                 278242, 278250, 279454, 274919,
                                 279537, 279239, 276777, 276668,
                                 274657, 279528, 271527, 278121,
                                 280724, 273789, 272840, 279078,
                                 273588, 285371, 286762, 282839,
                                 273329, 285019, 286771, 281810,
                                 271345, 279657, 289195, 282251,
                                 283563, 289886, 282394, 276670,
                                 280201, 289498, 279195, 289496,
                                 287752, 276619, 309938, 311405,
                                 311674, 310081, 314680, 315010,
                                 311968, 310975, 314710, 310924,
                                 310978, 315119, 310076, 311638,
                                 311131, 310641, 310803, 315362,
                                 315425, 310804, 315451, 311430,
                                 315200, 315180, 310610, 315396,
                                 315225, 312594, 312108, 312651,
                                 363746, 368125, 373722, 357038,
                                 380795, 392430, 387900
)
ORDER BY wp_header.wpno_i DESC
