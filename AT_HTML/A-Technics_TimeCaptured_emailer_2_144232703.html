                                       TO_CHAR(time_captured.scope) AS Тип_задач

                          LEFT JOIN wo_text_description ON workstep_link.descno_i = wo_text_description.descno_i
                          LEFT JOIN time_captured_additional ON time_captured.bookingno_i=time_captured_additional.bookingno_i
                          LEFT JOIN wo_text_action ON time_captured_additional.itemno_i=wo_text_action.actionno_i
                          LEFT JOIN workstep_link wsl_action ON wo_text_action.workstep_linkno_i=wsl_action.workstep_linkno_i
                          LEFT JOIN wo_text_description wtd_action ON wsl_action.descno_i = wtd_action.descno_i
                          JOIN wo_header ON time_captured.primkey = wo_header.event_perfno_i
                          JOIN wp_assignment ON wp_assignment.event_perfno_i=wo_header.event_perfno_i
                          JOIN wp_header ON wp_assignment.wpno_i=wp_header.wpno_i
                          LEFT JOIN aircraft ON aircraft.ac_registr = wo_header.ac_registr
                          LEFT JOIN SIGN ON time_captured.user_sign =sign.user_sign AND (sign.company_new = '0011026298' OR sign.department = '11026298')
                          LEFT JOIN wo_event_link ON wo_header.event_perfno_i=wo_event_link.event_perfno_i
                          WHERE 1=1
                                         AND wo_header.state <> 'V'
                                         AND wp_header.maintprov_address='0011026298' 
                                         AND (
                                                to_date('31.12.1971','DD.MM.YYYY')+wp_header.start_date BETWEEN trunc(sysdate-3) AND trunc(sysdate+3)
                                                OR
                                                (

                                                  AND 
                                                  (((wp_header.end_date is null)
                                                  OR 
                                                  (FLOOR(sysdate-to_date('19711231','yyyymmdd'))*(24*60) + TO_CHAR(SYSTIMESTAMP, 'SSSSS')/60 <= wp_header.end_date*24*60 + wp_header.end_time))))
                                                )
                                                OR
                                                (
                                                  ((FLOOR(sysdate-to_date('19711231','yyyymmdd'))*(24*60) + TO_CHAR(SYSTIMESTAMP, 'SSSSS')/60 >= wp_header.act_start_date*24*60 + wp_header.act_start_time)
                                                  AND 
                                                  (((wp_header.act_end_date is null)
                                                  OR

                                                )
                                         )
