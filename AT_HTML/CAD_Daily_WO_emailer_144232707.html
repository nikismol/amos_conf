SELECT wo_a.event_perfno_i as workorder,
             wo_a.state,
             wo_a.wo_type,
             wo_a.ata_chapter,
             wo_a.ac_typ as ac_type,
             wo_a.ac_registr,
             wo_a.created_date,
             wo_a.created_time,
             DECODE(wo_text_description.header,NULL, NULL, wo_text_description.header||CHR(10))
             ||wo_text_description.text||CHR(10)
             ||wo_text_description.text_html as wo_description,
             wo_a.action_text
FROM (
             SELECT wo.event_perfno_i,
                          wo.state,
                          DECODE(wo.type, 'C', 'Cabin', 'P', 'Pirep', 'M', 'Maintenance', '') as wo_type,
                          wo.ata_chapter,
                          wo.ac_typ,
                          wo.ac_registr,
                          wo.created_date,
                          wo.created_time,
                          RTRIM(
                                      XMLAGG(
                                                       XMLELEMENT(E,wo_text_action.text, CHR(10) ).EXTRACT('//text()')
                                                                                ORDER BY workstep_link.sequenceno, wo_text_action.actionno_i
                                                      ).GetClobVal(), CHR(10)
                                     ) as action_text
             FROM (
                          SELECT wo_header.event_perfno_i,
                                       wo_header.state,
                                       wo_header.type,
                                       aircraft.ac_typ,
                                       wo_header.ac_registr,
                                       wo_header.ata_chapter,
                                       TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + bohi_changes.changes_date, 'dd.mm.yyyy') as created_date,
                                       TO_CHAR(trunc(sysdate) + bohi_changes.changes_time/24/60, 'hh24:mi') as created_time
                          FROM wo_header
                          JOIN aircraft ON wo_header.ac_registr=aircraft.ac_registr
                          JOIN bohi_version ON bohi_version.primkey=wo_header.event_perfno_i AND bohi_version.mime_type = 'WO'
                          JOIN bohi_changes ON bohi_version.versionno_i=bohi_changes.versionno_i AND bohi_changes.action LIKE '%Manually entered W/O with Workorder 2%'
                          WHERE 1=1
                                        AND wo_header.created_date>=FLOOR(sysdate-to_date('19711231','yyyymmdd'))-1
                                        AND wo_header.state<>'V'
                                        AND bohi_changes.changes_date*(24*60)++bohi_changes.changes_time>= FLOOR(sysdate-to_date('19711231','yyyymmdd'))*(24*60) + TO_CHAR(SYSTIMESTAMP, 'SSSSS')/60 - 24*60
                                        AND wo_header.type IN ('P','C','M')
                         ) wo
             LEFT JOIN workstep_link ON wo.event_perfno_i=workstep_link.event_perfno_i
             LEFT JOIN wo_text_action ON wo_text_action.workstep_linkno_i=workstep_link.workstep_linkno_i
             GROUP BY wo.event_perfno_i,
                                wo.state,
                                wo.ac_registr,
                                wo.created_date,
                                wo.created_time,
                                wo.ac_typ,
                                wo.type,
                                wo.ata_chapter
           ) wo_a
LEFT JOIN workstep_link ON wo_a.event_perfno_i=workstep_link.event_perfno_i AND workstep_link.sequenceno=1
LEFT JOIN wo_text_description ON workstep_link.descno_i=wo_text_description.descno_i
ORDER BY 5, 6, 1
