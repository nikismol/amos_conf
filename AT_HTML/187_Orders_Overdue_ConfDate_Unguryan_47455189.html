Рассылка настроена в рамках TQ 42521.

Деактивирована

Результат: Осуществляется автоматическая отправка отчета "187_Orders_Overdue_ConfDate_Unguryan".

Разработчик:  Иваненко Владимир Андреевич.

SELECT 
     od_header.orderno AS OrderNumber
     ,od_header.ext_state AS OrderStatus
     ,od_header.created_by AS Creator
     ,sign.lastname||' '||sign.firstname AS Name
     ,od_header.ext_orderno AS ExternalOrderNumber
     ,od_detail.posno AS PositionNumber
     ,od_detail.ext_state AS DetailState
     ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (od_header.orderdate), 'DD.MM.YYYY')  AS OrderDate
     ,od_header.vendor AS VendorAddress
     ,address.name AS VendorName
     ,od_detail.partno AS PartNumber
     ,od_detail.serialno AS SerialNumber
     ,od_detail.labelno AS LabelNumber
     ,part.description AS PartDescription
     ,od_detail.qty AS QTY
     ,od_detail.backorder AS BACKORDER
     ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (od_detail.target_date), 'DD.MM.YYYY') AS TargetDate
     ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (od_detail.confirmed_date), 'DD.MM.YYYY') AS ConfirmedDate
FROM od_header
LEFT JOIN od_detail ON od_header.orderno_i = od_detail.orderno_i
LEFT JOIN part ON od_detail.partno = part.partno
LEFT JOIN address ON od_header.vendor = address.vendor
LEFT JOIN sh_detail ON od_detail.detailno_i = sh_detail.od_detailno_i
LEFT JOIN sh_header ON sh_detail.shipmentno_i = sh_header.shipmentno_i
LEFT JOIN part_special ON part.partno =  part_special.partno
LEFT JOIN sign ON od_header.created_by = sign.user_sign
WHERE TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (GREATEST(od_detail.target_date, od_detail.confirmed_date)), 'DD.MM.YYYY') < TO_DATE(sysdate)
  AND od_detail.ext_state = 'O'
  AND od_header.order_type = 'P'
  AND od_detail.confirmed_date IS NOT NULL 
  AND TO_CHAR(ADD_MONTHS(SYSDATE, -1), 'MM.YYYY (month)') = TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + od_detail.confirmed_date, 'MM.YYYY (month)')
  AND od_detail.confirmed_date <> '0' 
  AND part_special.special IN  ('PR012', 'PR001')
  AND NVL(sh_header.shipmentno_i, 1) = 1
ORDER BY od_detail.confirmed_date DESC, od_header.orderno
