select od_header.orderno
    ,od_header.ext_state
    ,od_header.ext_orderno
    ,od_detail.posno
    ,od_detail.ext_state
    ,TO_CHAR(TO_DATE('01.01.1970', 'dd.Mm.yyyy') + (od_header.orderdate+729), 'dd.Mm.yyyy') as issue_date
    ,TO_CHAR(TO_DATE('01.01.1970', 'dd.Mm.yyyy') + (od_header.entry_date+729), 'dd.Mm.yyyy') as created_date
    ,od_header.vendor
    ,od_detail.partno
    ,part.mat_class
    ,od_detail.serialno
    ,ps.special
    ,pat.ac_typ
    ,od_detail.labelno
    ,od_detail.qty
    ,od_detail.backorder
    ,TO_CHAR(TO_DATE('01.01.1970', 'dd.Mm.yyyy') + (od_detail.target_date+729), 'dd.Mm.yyyy') as target_date
    ,CASE WHEN od_detail.confirmed_date = 0 THEN ' ' ELSE TO_CHAR(TO_DATE('01.01.1970', 'dd.Mm.yyyy') + (od_detail.confirmed_date+729), 'dd.Mm.yyyy') END as confirmed_date
    ,od_detail.priority
    ,od_header.address_issue
    ,od_header.address_ship
    ,od_detail.ac_registr
    ,od_detail.reminder_count
    ,CASE WHEN od_detail.reminder_date = 0 THEN ' ' ELSE TO_CHAR(TO_DATE('01.01.1970', 'dd.Mm.yyyy') + (od_detail.reminder_date+729), 'dd.Mm.yyyy') END as reminder_date
    ,od_header.created_by
from part
left join ( 
        select 
                part_special.partno
                , LISTAGG (part_special.special, ';') WITHIN GROUP (ORDER BY part_special.partno) as special
        from part_special
        group by part_special.partno
) ps on part.partno = ps.partno
left join (
        select part_ac_type.partno,
               listagg(part_ac_type.ac_typ, ', ') within group (order by part_ac_type.ac_typ) as ac_typ
        from part_ac_type
        group by part_ac_type.partno
) pat on pat.partno = part.partno
left join od_detail on part.partno = od_detail.partno
left join od_header on od_detail.orderno_i = od_header.orderno_i
left join sh_detail on od_detail.detailno_i = sh_detail.od_detailno_i
left join sh_header on sh_detail.shipmentno_i = sh_header.shipmentno_i
where od_detail.ext_state in ('O','PR')
and od_header.order_type = 'P'
and od_header.vendor not in ('TECH_V', 'DUZD_TECH')
