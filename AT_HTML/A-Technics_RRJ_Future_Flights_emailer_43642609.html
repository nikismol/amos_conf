Результат запроса возвращает список TQs, обработанных специалистами RAMAX за период с 1 по 22 число

SELECT
ext_system_leg_id
,fn_suffix
,ssim_flight_identifier
,legno_i
,ac_registr
,fn_carrier
,fn_number
,service_type
,sched_departure_date
,sched_departure_time
,estim_departure_date
,estim_departure_time
,sched_departure_airport
,sched_arrival_date
,sched_arrival_time
,estim_arrival_date
,estim_arrival_time
,sched_arrival_airport
,estim_leg_duration
,leg_cycles
,sched_tah_at_arrival
,sched_tac_at_arrival
,mutation
,mutator
,status
,mutation__time
,created_by
,created_date
FROM (
             SELECT *
             FROM ( 
                          SELECT
                                         1 as head,
                                         TO_CHAR(ac_future_flights.ext_system_leg_id) as ext_system_leg_id,
                                         TO_CHAR(ac_future_flights.fn_suffix) as fn_suffix,
                                         TO_CHAR(ac_future_flights.ssim_flight_identifier) as ssim_flight_identifier, 
                                         TO_CHAR( ac_future_flights.legno_i) as legno_i,
                                         ac_future_flights.ac_registr,
                                         TO_CHAR(ac_future_flights.fn_carrier) as fn_carrier,
                                         TO_CHAR(ac_future_flights.fn_number) as fn_number,
                                         ac_future_flights.service_type,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_future_flights.sched_departure_date, 'DD.MM.YYYY') as sched_departure_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.sched_departure_time)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.sched_departure_time),60)),'FM00') as sched_departure_time,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_future_flights.estim_departure_date, 'DD.MM.YYYY') as estim_departure_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.estim_departure_time)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.estim_departure_time),60)),'FM00') as estim_departure_time,
                                         ac_future_flights.sched_departure_airport,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_future_flights.sched_arrival_date, 'DD.MM.YYYY') as sched_arrival_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.sched_arrival_time)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.sched_arrival_time),60)),'FM00') as sched_arrival_time,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_future_flights.estim_arrival_date, 'DD.MM.YYYY') as estim_arrival_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.estim_arrival_time)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.estim_arrival_time),60)),'FM00') as estim_arrival_time,
                                         ac_future_flights.sched_arrival_airport,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.estim_leg_duration)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.estim_leg_duration),60)),'FM00') as estim_leg_duration,
                                         TO_CHAR(TO_NUMBER(ac_future_flights.leg_cycles)) as leg_cycles,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.sched_tah_at_arrival)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.sched_tah_at_arrival),60)),'FM00') as sched_tah_at_arrival,
                                         TO_CHAR(TO_NUMBER(ac_future_flights.sched_tac_at_arrival)) as sched_tac_at_arrival,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_future_flights.mutation, 'DD.MM.YYYY') as mutation,
                                         ac_future_flights.mutator,
                                         TO_CHAR(ac_future_flights.status) as status,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_future_flights.mutation_time/1000/60)/60),'FM9900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_future_flights.mutation_time/1000/60),60)),'FM00') as mutation__time,
                                         ac_future_flights.created_by,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_future_flights.created_date, 'DD.MM.YYYY') as created_date
                          FROM ac_future_flights
                          JOIN aircraft ON ac_future_flights.ac_registr = aircraft.ac_registr AND aircraft.ac_typ='RRJ'
                          WHERE (ac_future_flights.sched_departure_date= trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy')
                                        OR ac_future_flights.sched_departure_date = trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy') + 1
                                        OR ac_future_flights.sched_arrival_date = trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy')
                                        OR ac_future_flights.sched_arrival_date = trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy') - 1
                                        OR ac_future_flights.estim_arrival_date = trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy')
                                        OR ac_future_flights.estim_arrival_date = trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy') - 1
                                        OR ac_future_flights.estim_departure_date = trunc(sysdate) -  to_date('31.12.1971', 'dd.mm.yyyy')
                                        OR ac_future_flights.estim_departure_date= trunc(sysdate) - to_date('31.12.1971', 'dd.mm.yyyy') + 1
                                       )

                                         AND (aircraft.maintenance_provider='0011026298'
                                                    OR aircraft.ac_registr IN (
                                                                                                   '89022',
                                                                                                   '89023',
                                                                                                   '89025',
                                                                                                   '89028',
                                                                                                   '89042',
                                                                                                   '89043',
                                                                                                   '89046',
                                                                                                   '89047',
                                                                                                   '89057',
                                                                                                   '89058',
                                                                                                   '89085',
                                                                                                   '89097',
                                                                                                   '89099',
                                                                                                   '89100',
                                                                                                   '89102',
                                                                                                   '89103',
                                                                                                   '89104',
                                                                                                   '89105',
                                                                                                   '89106',
                                                                                                   '89107',
                                                                                                   '89108',
                                                                                                   '89109',
                                                                                                   '89110',
                                                                                                   '89111',
                                                                                                   '89112',
                                                                                                   '89113',
                                                                                                   '89114',
                                                                                                   '89115',
                                                                                                   '89115',
                                                                                                   '89116')
                                                  )

                          UNION ALL

                          SELECT 2 as head,
                                         'ext_system_leg_id' as ext_system_leg_id,
                                         'fn_suffix' as fn_suffix,
                                         'ssim_flight_identifier' as ssim_flight_identifier,
                                         'legno_i' as legno_i,
                                         'ac_registr' as ac_registr,
                                         'fn_carrier' as fn_carrier,
                                         'fn_number' as fn_number,
                                         'service_type' as service_type,
                                         'sched_departure_date' as sched_departure_date,
                                         'sched_departure_time' as sched_departure_time,
                                         'estim_departure_date' as estim_departure_date,
                                         'estim_departure_time' as estim_departure_time,
                                         'sched_departure_airport' as sched_departure_airport,
                                         'sched_arrival_date' as sched_arrival_date,
                                         'sched_arrival_time' as sched_arrival_time,
                                         'estim_arrival_date' as estim_arrival_date,
                                         'estim_arrival_time' as estim_arrival_time,
                                         'sched_arrival_airport' as sched_arrival_airport,
                                         'estim_leg_duration' as estim_leg_duration,
                                         'leg_cycles' as leg_cycles,
                                         'sched_tah_at_arrival' as sched_tah_at_arrival,
                                         'sched_tac_at_arrival' as sched_tac_at_arrival,
                                         'mutation' as mutation,
                                         'mutator' as mutator,
                                         'status' as status,
                                         'mutation_time' as mutation__time,
                                         'created_by' as created_by,
                                         'created_date' as created_date
                          FROM DUAL
                          ORDER BY 1 DESC
)
)
