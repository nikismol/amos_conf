FROM (
             SELECT *
             FROM (
                          SELECT 2 as head,

                                         TO_CHAR(ac_utilization.fn_suffix) as fn_suffix,
                                         TO_CHAR(ac_utilization.legno_i) as legno_i,
                                         TO_CHAR(ac_utilization.ac_registr) as ac_registr, 
                                         TO_CHAR(ac_utilization.validity) as validity,
                                         TO_CHAR(ac_utilization.recno) as recno,
                                         TO_CHAR(ac_utilization.null_leg_commited) as null_leg_commited,
                                         TO_CHAR(ac_utilization.leg_number) as leg_number,
                                         TO_CHAR(ac_utilization.fn_carrier) as fn_carrier,
                                         TO_CHAR(ac_utilization.fn_number) as fn_number,
                                         TO_CHAR(ac_utilization.service_type) as service_type,
                                         TO_CHAR(ac_utilization.netline_leg_id) as netline_leg_id,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_utilization.departure_date, 'DD.MM.YYYY') as departure_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.departure_time)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.departure_time),60)),'FM00') as departure_time,
                                         TO_CHAR(ac_utilization.departure) as departure,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_utilization.sched_departure_date, 'DD.MM.YYYY') as sched_departure_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.sched_departure_time)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.sched_departure_time),60)),'FM00') as sched_departure_time,
                                         TO_CHAR(ac_utilization.sched_departure_airport) as sched_departure_airport,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.arrival_time)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.arrival_time),60)),'FM00') as arrival_time,
                                         TO_CHAR(ac_utilization.arrival) as arrival,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_utilization.sched_arrival_date, 'DD.MM.YYYY') as sched_arrival_date,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.sched_arrival_time)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.sched_arrival_time),60)),'FM00') as sched_arrival_time,
                                         TO_CHAR(ac_utilization.sched_arrival_airport) as sched_arrival_airport,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.daily_hours)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.daily_hours),60)),'FM00') as daily_hours,
                                         TO_CHAR(TO_NUMBER(ac_utilization.daily_cycles)) as daily_cycles,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.tah)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.tah),60)),'FM00') as tah,
                                         TO_CHAR(ac_utilization.tac) as tac,
                                         TO_CHAR(ac_utilization.special_flight_type) as special_flight_type,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_utilization.mutation, 'DD.MM.YYYY') as mutation,
                                         TO_CHAR(ac_utilization.mutator) as mutator,
                                         TO_CHAR(ac_utilization.status) as status,
                                         TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(ac_utilization.mutation_time)/60),'FM999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(ac_utilization.mutation_time),60)),'FM00') as mutation_time,
                                         TO_CHAR(ac_utilization.created_by) as created_by,
                                         TO_CHAR(to_date('31.12.1971','DD.MM.YYYY')+ac_utilization.created_date, 'DD.MM.YYYY') as created_date
                          FROM ac_utilization
                          JOIN aircraft ON ac_utilization.ac_registr = aircraft.ac_registr AND aircraft.ac_typ='RRJ'
                          WHERE to_date('31.12.1971', 'dd.mm.yyyy') + ac_utilization.validity =trunc(sysdate)
                                         AND (aircraft.maintenance_provider='0011026298' OR aircraft.ac_registr IN (
                                                                                                                                                                                        '89022',
                                                                                                                                                                                        '89023',
                                                                                                                                                                                        '89025',
                                                                                                                                                                                        '89028',
                                                                                                                                                                                        '89042',

                                                                                                                                                                                        '89115',
                                                                                                                                                                                        '89116')
                                                 )

                          UNION ALL

                          SELECT 1 as head,
                                         'ext_system_leg_id' as ext_system_leg_id,
                                         'fn_suffix' as fn_suffix,
                                         'legno_i' as legno_i,
                                         'ac_registr' as ac_registr,
                                         'validity' as validity,
                                         'recno' as recno,
                                         'null_leg_commited' as null_leg_commited,
                                         'leg_number' as leg_number,
                                         'fn_carrier' as fn_carrier,
                                         'fn_number' as fn_number,
                                         'service_type' as service_type,
                                         'netline_leg_id' as netline_leg_id,
                                         'departure_date' as departure_date,
                                         'departure_time' as departure_time,
                                         'departure' as departure,
                                         'sched_departure_date' as sched_departure_date,
                                         'sched_departure_time' as sched_departure_time,
                                         'sched_departure_airport' as sched_departure_airport,
                                         'arrival_time' as arrival_time,
                                         'arrival' as arrival,
                                         'sched_arrival_date' as sched_arrival_date,
                                         'sched_arrival_time' as sched_arrival_time,
                                         'sched_arrival_airport' as sched_arrival_airport,
                                         'daily_hours' as daily_hours,
                                         'daily_cycles' as daily_cycles,
                                         'tah' as tah,
                                         'tac' as tac,
                                         'special_flight_type' as special_flight_type,
                                         'mutation' as mutation,
                                         'mutator' as mutator,
                                         'status' as status,
                                         'mutation_time' as mutation_time,
                                         'created_by' as created_by,
                                         'created_date' as created_date
                          FROM DUAL
                          ORDER BY 1
)
