AMOS Team : Получение сертификата с Web Drive и локальной папки

Created by Филипенков Никита Константинович, last modified on сент. 28, 2023 
SELECT
    *
FROM (
    SELECT
        r.partno,
        r.serialno,
        h.voucherno || ' ' || 'см в APN 188' AS doc,
        'web://'||cert_link.destination_key AS path
    FROM rotables r
    LEFT JOIN history h ON r.psn=h.psn AND h.VM='B1'
    LEFT JOIN db_link cert_link ON h.recdetailno_i=cert_link.source_pk AND cert_link.source_type IN ('RD', 'HI') AND cert_link.link_type='D'
    LEFT JOIN location ON r.locationno_i=LOCATION.LOCATIONNO_I
    WHERE location.station='SHRM'
      AND location.store='1221'
      AND cert_link.LINKNO_I is not null
    UNION
    SELECT
        r.partno,
        r.serialno,
        'form tracking no:' ||' '|| cert.form_tracking_no ,
        '\\msk.aeroflot.ru\amos\PROD\WEB_DRIVE\'||ste_content_list.engine_id AS path
    FROM rotables r
    LEFT JOIN certificate_link cert_link ON r.psn=cert_link.source_key
    LEFT JOIN certificate cert ON cert.certificate_i = cert_link.certificate_i
    LEFT JOIN db_link ON db_link.source_type = 'CERT' AND db_link.source_pk = cert.certificate_i AND db_link.link_type = 'S'
    LEFT JOIN ste_content_list ON ste_content_list.storage_id = db_link.destination_key
    LEFT JOIN location ON r.locationno_i=location.locationno_i
    WHERE location.station='SHRM'
      AND location.store='1221'
      AND cert_link.source_type = 'RO'
      AND ste_content_list.engine_id IS NOT NULL
    )
ORDER BY partno, serialno
