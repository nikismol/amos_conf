AMOS Team : Получение TBI и CBI для Rotables

Created by Филипенков Никита Константинович on авг. 18, 2023 
SELECT
    TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(history.mbi )/60),'FM999999999900')) || ':'
                            ||TO_CHAR(TRUNC(MOD(TO_NUMBER(history.mbi ),60)),'FM00') as tbi,
    TO_CHAR(TO_NUMBER(history.cbi)) as cbi
FROM history
