SELECT
    history.partno PART_NUMBER,
    history.serialno SERIAL_NUMBER,
    history.batchno BATCH_NUMBER,
    part.description PART_DESCRIPTION,
    history.owner OWNER,
    history.ata_chapter ATA_CHAPTER,
    history.mat_class MAT_CLASS,
    part.mat_type MAT_TYPE,
    CASE
        WHEN history.vm='T' AND history.voucherno like 'T-%' THEN history_text.station
        ELSE history.station
    END STATION,
    CASE
        WHEN history.vm='T' AND history.voucherno like 'T-%' THEN history_text.store
        ELSE history.store
    END STORE,
    CASE
        WHEN history.vm='T' AND history.voucherno like 'T-%' THEN history_text.location
        ELSE history.location
    END LOCATION,
    history.vm VOUCHER_MODE,
    history.voucherno VOUCHER_NUMBER,
    history.ac_registr AC_REGISTR,
    history.receiver RECEIVER,
    TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (history.del_date), 'DD.MM.YYYY') BOOKING_DATE,
    history.qty QTY,
    part.measure_unit MEASURE_UNIT,
    CASE
        WHEN history.vm='T' AND history.voucherno like 'T-%' THEN history.station
        ELSE pickslip_header.station_to
    END STATION_TO,
    CASE
        WHEN history.vm='T' AND history.voucherno like 'T-%' THEN history.store
        ELSE pickslip_header.store_to
    END STORE_TO,
    history.average_price SELECTED_PRICE_RUB
FROM history
LEFT JOIN (
    SELECT
        REGEXP_SUBSTR(history_text.short_text, ':\s*([^:/]+)', 1, 1, NULL, 1) AS station,
        REGEXP_SUBSTR(history_text.short_text, '/([^/]+)/', 1, 1, NULL, 1) AS store,
        REGEXP_SUBSTR(history_text.short_text, '([^:/]+)$', 1, 1) AS location,
        history_text.historyno_i
    FROM history_text
    WHERE history_text.short_text LIKE 'Previous Location:%'
) history_text ON history.historyno_i=history_text.historyno_i
LEFT JOIN part ON history.partno = part.partno
LEFT JOIN od_detail ON history.od_detailno_i = od_detail.detailno_i
LEFT JOIN pickslip_booked ON history.pickslipseqno_i = pickslip_booked.pickslipseqno_i
LEFT JOIN pickslip_header ON pickslip_booked.pickslipno = pickslip_header.pickslipno
LEFT JOIN project ON pickslip_header.projectno = project.projectno
LEFT JOIN location ON history.location = location.location AND ( history.store = location.store AND history.station = location.station )
WHERE (
    history.del_date between FLOOR(trunc(ADD_MONTHS(SYSDATE, -1), 'MM')-to_date('19711231','yyyymmdd'))
        and
        FLOOR(trunc(SYSDATE, 'MM') -1-to_date('19711231','yyyymmdd'))
    )
  AND history.station||history.store IN (
                                         'SVOHANGAR_4',
                                         'SVOHANGAR_5',
                                         'SVOAT_ACC',
                                         'SVOAT_BASE',
                                         'SVOAT_CABIN',
                                         'SVOAT_FLUID',
                                         'SVOAT_WHEEL',
                                         'SHRM5010'
                                        )
