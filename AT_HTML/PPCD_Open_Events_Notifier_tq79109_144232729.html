SELECT ac_typ,
              ac_registr,
              station,
              wpno,
              wp_end_at,
              event_no,
              event_type,
              event_state,
              flight_number,
              departure_at
FROM (
             SELECT wp_header.ac_typ,
                           wp_header.ac_registr,
                           wp_header.wpno,
                           wp_header.station,
                           CASE WHEN wo_header.workorderno_display > 0 THEN TO_CHAR(wo_header.workorderno_display) ELSE event_name END as event_no,
                           CASE WHEN wo_header.workorderno_display > 0 THEN 'WO' ELSE 'TC' END as event_type,
                           'Open' as event_state,
                           ac_utilization.fn_number as flight_number,
                           DENSE_RANK() over(PARTITION BY ac_utilization.ac_registr ORDER BY ac_utilization.departure_date*24*60 + ac_utilization.departure_time ASC) as num,
                           TO_CHAR(TO_DATE('31.12.1971','dd.mm.yyyy') + wp_header.end_date, 'dd.mm.yyyy') ||' '|| to_char(trunc(sysdate) + interval '1' minute * wp_header.end_time, 'hh24:mi') as wp_end_at,
                           TO_CHAR(TO_DATE('31.12.1971','dd.mm.yyyy') + ac_utilization.departure_date, 'dd.mm.yyyy') ||' '|| to_char(trunc(sysdate) + interval '1' minute * ac_utilization.departure_time, 'hh24:mi') as departure_at
             FROM wp_header
             JOIN aircraft ON wp_header.ac_registr = aircraft.ac_registr AND aircraft.status <> 9
             JOIN wp_assignment ON wp_header.wpno_i = wp_assignment.wpno_i
             JOIN wo_header ON wp_assignment.event_perfno_i = wo_header.event_perfno_i
             LEFT JOIN wo_event_link ON wo_header.event_perfno_i = wo_event_link.event_perfno_i AND wo_event_link.event_type = 'TCEFFLI'
             JOIN ac_utilization ON wp_header.ac_registr = ac_utilization.ac_registr
                                                      AND ac_utilization.departure_date*24*60 + ac_utilization.departure_time >= wp_header.end_date*24*60 + wp_header.end_time
                                                      AND ac_utilization.recno > 0
             WHERE 1=1
                            AND wp_header.wp_status IN (-1, -2, -3)
                            AND (wo_header.workorderno_display > 0 OR wo_event_link.event_perfno_i > 0)
                            AND wo_header.state IN ('O')
             ORDER BY 1, 2, 3
            )
WHERE 1=1
               AND num = 1
               AND (EXTRACT(DAY FROM (SYSTIMESTAMP AT TIME ZONE 'UTC' - TO_TIMESTAMP_TZ(departure_at||' GMT', 'DD.MM.YYYY HH24:MI TZR'))) * 24) +
                         EXTRACT(HOUR FROM (SYSTIMESTAMP AT TIME ZONE 'UTC' - TO_TIMESTAMP_TZ(departure_at||' GMT', 'DD.MM.YYYY HH24:MI TZR'))) >= 15
