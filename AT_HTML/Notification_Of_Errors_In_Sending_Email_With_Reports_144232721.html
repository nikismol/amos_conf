SELECT
   to_char(workflow_rule.workflowno_i) AS workflowno,
   to_char(workflow_rule.rule_name) AS rule_name,
   to_char(workflow_log_message.message_description) AS message_description,
   to_char(workflow_log_message.error_message) AS error_message,
   to_char(workflow_log_action.properties) AS properties
FROM workflow_rule
LEFT JOIN workflow_log_rule ON workflow_rule.RULENO_I=workflow_log_rule.workflow_rule_id
LEFT JOIN workflow_log_message ON workflow_log_rule.workflow_rule_log_id=workflow_log_message.workflow_log_rule
LEFT JOIN workflow_log_action ON workflow_log_rule.workflow_rule_log_id=workflow_log_action.workflow_log_action_id
WHERE workflow_rule.workflowno_i = 12
  AND workflow_rule.rule_name='MELReport_emailer_2'
  AND workflow_log_message.workflow_log_message_id IS NOT NULL
  AND workflow_log_rule.date_executed = (TRUNC(sysdate) - TO_DATE('31.12.1971', 'DD.MM.YYYY'))
  AND (
      workflow_log_rule.date_executed >= ((TO_NUMBER(TO_CHAR(sysdate, 'HH24')) * 60) + TO_NUMBER(TO_CHAR(sysdate, 'MI'))-10)
          AND
      workflow_log_rule.date_executed <= ((TO_NUMBER(TO_CHAR(sysdate, 'HH24')) * 60) + TO_NUMBER(TO_CHAR(sysdate, 'MI')))
      )
UNION
SELECT
    'EXP_CHK_OK',
    null,
    null,
    null,
    null
FROM dual
WHERE (
    SELECT count(1)
    FROM workflow_rule
    LEFT JOIN workflow_log_rule ON workflow_rule.RULENO_I=workflow_log_rule.workflow_rule_id
    LEFT JOIN workflow_log_message ON workflow_log_rule.workflow_rule_log_id=workflow_log_message.workflow_log_rule
    LEFT JOIN workflow_log_action ON workflow_log_rule.workflow_rule_log_id=workflow_log_action.workflow_log_action_id
    WHERE workflow_rule.workflowno_i = 12
      AND workflow_rule.rule_name='MELReport_emailer_2'
      AND workflow_log_message.workflow_log_message_id IS NOT NULL
      AND workflow_log_rule.date_executed = (TRUNC(sysdate) - TO_DATE('31.12.1971', 'DD.MM.YYYY'))
      AND (
          workflow_log_rule.date_executed >= ((TO_NUMBER(TO_CHAR(sysdate, 'HH24')) * 60) + TO_NUMBER(TO_CHAR(sysdate, 'MI'))-10)
