SELECT ac_typ,
              ac_registr,
              wpno,
              station,
              wp_end_date_time,
              flight_number,
              departure_at
FROM (
             SELECT wp_header.ac_typ,
                           wp_header.ac_registr,
                           wp_header.wpno,
                           wp_header.station,
                           TO_CHAR(TO_DATE('31.12.1971','dd.mm.yyyy') + wp_header.end_date, 'dd.mm.yyyy') ||' '|| to_char(trunc(sysdate) + interval '1' minute * wp_header.end_time, 'hh24:mi') as wp_end_date_time,
                           ac_utilization.fn_number as flight_number,
                           TO_CHAR(TO_DATE('31.12.1971','dd.mm.yyyy') + ac_utilization.departure_date, 'dd.mm.yyyy') 
                           || ' ' || to_char(trunc(sysdate) + interval '1' minute * ac_utilization.departure_time, 'hh24:mi') as departure_at,
                           row_number() over(PARTITION BY ac_utilization.ac_registr ORDER BY ac_utilization.departure_date*24*60 + ac_utilization.departure_time ASC) as num
             FROM wp_header
             JOIN ac_utilization ON wp_header.ac_registr = ac_utilization.ac_registr
                                                      AND wp_header.end_date*24*60 + wp_header.end_time < ac_utilization.departure_date*24*60 + ac_utilization.departure_time
                                                      AND ac_utilization.recno > 0
             WHERE 1=1
                            AND wp_header.wp_status IN (-1, -3) 
                            AND wp_header.end_date >= 18629
                            AND act_start_date IS NULL
                            AND act_end_date IS NULL
                            AND wp_header.station NOT IN ('ZZZ', 'AAA', 'RRR', 'LLL', 'SSS')
                            AND (EXTRACT(DAY FROM (
                                                                             SYSTIMESTAMP AT TIME ZONE 'UTC' - TO_TIMESTAMP_TZ(TO_CHAR(TO_DATE('31.12.1971','dd.mm.yyyy') + ac_utilization.departure_date, 'dd.mm.yyyy')
                                                                             ||' '
                                                                             || to_char(trunc(sysdate) + interval '1' minute * ac_utilization.departure_time, 'hh24:mi')
                                                                             ||' GMT', 'DD.MM.YYYY HH24:MI TZR'))) * 24)
                                     +
                                     EXTRACT(HOUR FROM (
                                                                               SYSTIMESTAMP AT TIME ZONE 'UTC' - TO_TIMESTAMP_TZ(TO_CHAR(TO_DATE('31.12.1971','dd.mm.yyyy') + ac_utilization.departure_date, 'dd.mm.yyyy')
                                                                               ||' '
                                                                               || to_char(trunc(sysdate) + interval '1' minute * ac_utilization.departure_time, 'hh24:mi')
                                                                               ||' GMT', 'DD.MM.YYYY HH24:MI TZR')))
                                     >= 4
             )
WHERE 1=1
              AND num = 1
