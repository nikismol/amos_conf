SELECT
    r.partno,
    r.serialno,       
    part.description,
    part.mat_class,
    location.station,
    location.store,
    location.location,
    TO_CHAR(TO_DATE('01.01.1970', 'dd.mm.yyyy') + (forecast.expected_date+729),'dd.mm.yyyy') AS expiry_date
FROM part
JOIN rotables r ON part.partno = r.partno
JOIN location ON r.locationno_i = location.locationno_i
JOIN forecast ON r.psn = forecast.psn AND r.ac_registr IN ('COMP','TRANSF','DUMMYR', ' ')
JOIN forecast_dimension ON forecast.event_perfno_i = forecast_dimension.event_perfno_i
LEFT JOIN wo_event_link ON forecast.event_perfno_i = wo_event_link.event_perfno_i
LEFT JOIN part_req ON wo_event_link.event_key_parent = part_req.reqseqno_i AND wo_event_link.event_type = 'RORE' 
                          AND wo_event_link.pending_status = 0 AND wo_event_link.planable_status = 0
WHERE forecast_dimension.dimension IN ('D', 'SD')
  AND forecast.event_type IN ('R')
  AND location.location_type <> -3
  AND r.locationno_i <> -200
  AND r.rotable_status <> 2
  AND r.partno IN
      (
       '3800702-1',
       '3800708-1',
       '3800454-6',
       '3800550-1',
       'WE3800770-3',
       'SAM146-1S17',
       'GE90-115B',
       'CFM56-7B26EG05',
       'CFM56-7B26',
       'FK30858',
       '1887M10G05',
       '1887M10G06',
       '1887M10G07'
      )
  AND  forecast.expected_date =FLOOR(SYSDATE-TO_DATE('19711231','yyyymmdd'))+30
