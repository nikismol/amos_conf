Результат: Информирует о том, что интерфейс SAP отработал с ошибками/предупреждениями. 

SELECT
    DECODE(scheduler_log.status,
        '1', 'INTERFACE_WARNING',
        '2', 'INTERFACE_ERROR',
        '7', 'INTERFACE_PARALLEL_EXECUTION',
        '6', 'EXECUTING',
        '0', 'SUCCESS')  /*6,0 for information only, not selected*/ as state,
    scheduler_conf.name,
    scheduler_log.status,
    scheduler_log.log_text
FROM scheduler_log
LEFT JOIN scheduler_conf ON scheduler_conf.scheduler_confno_i = scheduler_log.scheduler_confno_i
WHERE scheduler_log.mutation*(24*60*60*1000)+scheduler_log.mutation_time >= FLOOR(sysdate-to_date('19711231','yyyymmdd'))*(24*60*60*1000) + TO_CHAR(SYSTIMESTAMP, 'SSSSS')*1000-30*60*1000
  AND scheduler_log.scheduler_confno_i IN (
                                           2002,
                                           /*ExportComponentHistory*/ 2003,
                                           /*ExportAircraft*/ 6404,
                                           /*TransferPartDefition*/ 6402,
                                           /*TransferShipment*/ 2004,
                                           /*TransferOrder*/ 2005 /*TransferReceiving*/)
  AND scheduler_log.status !=0
  AND scheduler_log.status !=7
  AND scheduler_log.status !=6
