SELECT 
    CASE 
        WHEN dbms_lob.instr(scheduler_log.log_text, 'must not be later or equal to the')>0 OR dbms_lob.instr(scheduler_log.log_text, 'is not a valid value')>0 THEN 'ERR_EF_VALIDATION'  
        ELSE 'AOK' 
    END as sts, 
    scheduler_log.log_text as text 
FROM scheduler_log 
WHERE scheduler_log.scheduler_logno_i IN (
    SELECT 
        max(scheduler_log.scheduler_logno_i) 
    FROM scheduler_log 
    WHERE scheduler_log.scheduler_confno_i = 2604 
      AND status <> 6
    )
