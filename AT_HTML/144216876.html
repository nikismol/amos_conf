AMOS Team : Получение TSI и CSI для Rotables

Created by Филипенков Никита Константинович, last modified on авг. 18, 2023 
SELECT
    rotables.partno,
    rotables.serialno,
    rotables.ac_registr as AC_install,
    TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(hist.msn)/60),'FM999999999900')) || ':'
        ||TO_CHAR(TRUNC(MOD(TO_NUMBER(hist.msn),60)),'FM00') as tsn,
    hist.csn AS csn,
    TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(tam_ac.tam_ac-hist.tam)/60),'FM999999999900')) || ':'
        ||TO_CHAR(TRUNC(MOD(TO_NUMBER(tam_ac.tam_ac-hist.tam),60)),'FM00') as tsi,
    (tac_ac.tac_ac-hist.tac) AS csi
FROM rotables
FULL JOIN aircraft ON rotables.ac_registr=aircraft.ac_registr
LEFT JOIN (
    SELECT
        history.psn,
        max(history.historyno_i) AS historyno_i
    FROM history
    WHERE history.VM='YE'
    GROUP BY  history.psn
) history ON rotables.psn=history.psn
LEFT JOIN history hist ON history.historyno_i=hist.historyno_i
LEFT JOIN (
    SELECT
        counter.life_value as tac_ac,
        counter.ref_key
    FROM counter
    LEFT JOIN counter_template ON counter.counter_templateno_i=counter_template.counter_templateno_i
    LEFT JOIN counter_definition ON counter_template.counter_defno_i=counter_definition.counter_defno_i
    WHERE counter.ref_type='AC_INT'
      AND counter_definition.counter_defno_i='2'
) tac_ac on aircraft.aircraftno_i=tac_ac.ref_key
LEFT JOIN (
    SELECT
        counter.life_value as tam_ac,
        counter.ref_key
    FROM counter
    LEFT JOIN counter_template ON counter.counter_templateno_i=counter_template.counter_templateno_i
    LEFT JOIN counter_definition ON counter_template.counter_defno_i=counter_definition.counter_defno_i
    WHERE counter.ref_type='AC_INT'
      AND counter_definition.counter_defno_i='1'
) tam_ac on aircraft.aircraftno_i=tam_ac.ref_key
