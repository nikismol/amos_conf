SELECT 
    pm_publication.pm_publicationno_i as publication_no
    , pm_publication_type.type as publication_type
    , pm_publication.docno as document_no
    , TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (pm_publication.valid_until), 'DD.MM.YYYY') as evaluation_deadline 
    , pm_state_type.state_description || ' (' || pm_state_type.statetype || ')' as current_state
FROM 
    pm_publication 
LEFT JOIN 
    (
        SELECT
            max(ps.pm_stateno_i) as current_pm_stateno_i,
            ps.pm_publicationno_i
        FROM pm_state ps
        GROUP BY ps.pm_publicationno_i
    ) current_pm_state ON current_pm_state.pm_publicationno_i = pm_publication.pm_publicationno_i
LEFT JOIN pm_state ON pm_state.pm_stateno_i = current_pm_state.current_pm_stateno_i
LEFT JOIN pm_publication_type ON pm_publication.pm_typeno_i = pm_publication_type.pm_typeno_i
JOIN pm_state_type ON pm_state_type.pm_statetypeno_i = pm_state.pm_statetypeno_i AND (pm_state_type.statetype = 'RDEF' OR (pm_state_type.statetype = 'EVAL' AND pm_publication.valid_until<= FLOOR(sysdate-to_date('19711231','yyyymmdd')) ))
