SELECT 'activation_required' AS expire_check
FROM license_key
WHERE 1=1
  AND licenseno_i IN 
      (
      SELECT
          max(licenseno_i) 
      FROM license_key 
      WHERE status = 0) 
  AND TO_DATE('31.12.1971', 'dd.mm.yyyy') + created_date + 19 <= TRUNC(SYSDATE)
