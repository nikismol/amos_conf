SELECT
	TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (history.del_date), 'DD.MM.YYYY') AS utc_date, 
	part.description AS part_description, 
	history.partno AS partnumber, 
	history.serialno || history.batchno AS sn_bn, 
	part.measure_unit, 
	history.qty AS quantity, 
	history.mat_class, 
	CASE WHEN history.condition = `S` THEN `SERVICABLE / ИСПРАВНО` 
	WHEN history.condition = `US` THEN `UNSERVICABLE / НЕИСПРАВНО` 
	WHEN history.condition = `OH` THEN `OVERHAULED / КАПИТАЛЬНЫЙ РЕМОНТ` 
	WHEN history.condition = `N` THEN `NEW` 
	WHEN history.condition = `R` THEN `REPAIRED / ТЕКУЩИЙ РЕМОНТ`
	 ELSE history.condition 
	END condition,
	history.voucherno, 
	history.station ||` - ` || history.store AS receiver 
FROM history 
	LEFT JOIN part ON part.partno = history.partno 
WHERE 
	history.vm = `TS`
	AND history.qty < 0
	AND (
	history.del_date between FLOOR(trunc(ADD_MONTHS(SYSDATE, -1), 'MM')-to_date('19711231','yyyymmdd')) AND FLOOR(trunc(SYSDATE, 'MM') -1-to_date('19711231','yyyymmdd'))
                          )
	AND history.station||history.store IN (`SVOHANGAR_4`, `SVOHANGAR_5`, `SVOAT_ACC`, `SVOAT_BASE`, `SVOAT_CABIN`, `SVOAT_FLUID`, `SVOAT_WHEEL`, `SHRM5010`)
