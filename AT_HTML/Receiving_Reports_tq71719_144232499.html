SELECT od_rec_detail.receivingno,
       od_rec_detail.recdetailno_i,
       od_rec_detail.orderno,
       od_detail.posno,
       od_rec_detail.partno,
       od_rec_detail.condition,
       od_rec_detail.labelno,
       od_rec_header.awbno,
       od_detail.projectno,
       part.description,
       od_rec_detail.qty AS qty,
       od_rec_detail.mat_class,
       part.repairable,
       part.tool,
       ps.special,
       od_rec_detail.vendor,
       od_rec_detail.owner,
       history.del_date,
       history.station,
       history.store,
       history.location,
       od_header.address_bill,
       od_header.priority,
       TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (od_rec_detail.created_date), 'DD.MM.YYYY') as created_date,
       od_rec_detail.created_by,
       TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (od_rec_detail.mutation), 'DD.MM.YYYY') as mutation_date,
       floor(od_rec_detail.mutation_time / 3600000) || ':' || substr(mod(od_rec_detail.mutation_time, 3600000), 1, 2) as mutation_time,
       history.vm,
       od_rec_detail.serialno,
       od_rec_detail.psn,
       od_rec_detail.batchno,
       od_rec_detail.recno,
       od_rec_detail.ext_purch_pricec,
       od_rec_detail.currency
FROM od_rec_detail
JOIN history ON od_rec_detail.recdetailno_i = history.recdetailno_i
JOIN od_rec_header ON od_rec_detail.receivingno = od_rec_header.receivingno
JOIN part ON od_rec_detail.partno = part.partno
--left join part_special on part.partno = part_special.partno
LEFT JOIN ( 
        select 
                part_special.partno
                , LISTAGG (part_special.special, ';') WITHIN GROUP (ORDER BY part_special.partno) as special
        from part_special
        group by part_special.partno
) ps on part.partno = ps.partno
LEFT JOIN od_detail ON od_rec_detail.detailno_i = od_detail.detailno_i
LEFT JOIN od_header ON od_detail.orderno_i = od_header.orderno_i
LEFT JOIN inspection_detail ON inspection_detail.source_type = 'RD' AND inspection_detail.source_key = od_rec_detail.recdetailno_i OR (
             inspection_detail.source_type = 'ORDER_DE'
             AND inspection_detail.source_key = od_rec_detail.detailno_i
          ) OR (
             inspection_detail.source_type = 'LB'
             AND inspection_detail.source_key = od_detail.labelno
          )
WHERE history.vm IN (
         'B1',
         'B6',
         'B7',
         'B5'
      )
      AND od_rec_detail.status = 0
      AND od_header.order_type = 'P'
      AND history.del_date >= 18323
      AND history.del_date <= (FLOOR(TRUNC((sysdate)-to_date('31121970','ddmmyyyy'))))
