SELECT odh.orderno as ORDER_NUMBER
FROM od_header odh
JOIN od_detail odd ON odd.orderno_i = odh.orderno_i
WHERE odh.ext_state = `D`
    AND odd.partno IN 
    (SELECT DISTINCT partno FROM part_special WHERE special IN (`PR054`, `PR0564`, `PR083`, `PR084`))
