SELECT * FROM 
(
SELECT
to_number(queryno_i),
headline,
TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (created_date), 'DD.MM.YYYY') CREATED_DATE_1,
issued_by_group,
issued_by,
priority,
CASE WHEN (status!=3) THEN 'В работе' ELSE TO_CHAR(TO_DATE('31/12/1971', 'DD/MM/YYYY') + (mutation), 'DD/MM/YYYY') END AS CLOSURE_DATE
FROM tech_query
WHERE
queryno_i IN
(
SELECT distinct queryno_i FROM tech_query_note WHERE created_by IN
(
'FEDOSEEN',
'PASECHNI',
'LABUSHEV',
'IVANENKO',
'VRYBIN',
'NFILIP',
'AZORIKOV',
'ABELYKH' ,
'AKHAMZIN',
'AVOLKOVA',
'DENISP' ,
'GKOROLEV',
'SATDINOV',
'SKISELEV',
'SKOROLEV',
'VTSYMBAL',
'EFROLOVA',
'IKOKORIN',
'SIROTKIN'
)
)
ORDER BY queryno_i
)
WHERE CREATED_DATE_1 >= (to_date(sysdate)-21)
