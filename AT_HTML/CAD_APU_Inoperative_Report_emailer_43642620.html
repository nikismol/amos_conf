SELECT i.ac_registr as ac_registr,
	i.ac_typ as ac_typ,
	i.wono as wono,
	i.ata as ata,
	i.issue_date as issue_date,
               i.mel_code as mel_code,
               i.wo_due_date as wo_due_date,
               i.text as text,
TO_CHAR(CASE WHEN mel_detail.mel_detailno_i IS NULL
                              THEN ' '
                              ELSE '<b>MEL:</b><br />'||' <b>'||DECODE(mel_detail.ops_proc_req||mel_detail.maint_proc_req, 'YY', '(O)(M)', 'YN', '(O)', 'NY', '(M)', '')||'</b> '||TO_CHAR(mel_tree.mel_text)
                    END) as c_mel_text,
TO_CHAR(CASE WHEN ac_lim.event_perfno_i IS NULL
                             THEN '' ELSE ac_lim.limitations_agg
                   END) as limitations_agg_na,
TO_CHAR(CASE WHEN wo_opus.free_text IS NOT NULL
                             THEN TO_CLOB(CONCAT(TO_CLOB('<br /><b>Comments:</b><br />'), TO_CLOB(wo_opus.free_text)))
                             ELSE TO_CLOB('')
                   END) as mcc_remarks,
TO_CHAR(
                   CASE WHEN mel_detail.mel_detailno_i IS NOT NULL
                              THEN CASE WHEN aircraft.ac_typ IN ('A32S', 'A330', 'RRJ')
                                                    THEN TO_CLOB(
                                                                                  CONCAT(TO_CLOB('<b>MEL Remarks:</b><br />'), mel_detail.mel_remarks)
                                                                                )
                                                    ELSE CASE WHEN LENGTH(mel_procedure.maint_proc_text) > 1 OR LENGTH(mel_procedure.ops_proc_text) > 1
                                                                        THEN TO_CLOB(
                                                                                                      CONCAT(
                                                                                                                       TO_CLOB('<br /><b>Procedures:</b><br />'), 						
                                                                                                                       TO_CLOB(
                                                                                                                                         CONCAT(
                                                                                                                                                          CASE WHEN LENGTH(mel_procedure.ops_proc_text) > 1
                                                                                                                                                                     THEN TO_CLOB(CONCAT(TO_CLOB('<br /><b>(O)</b><br />'), mel_procedure.ops_proc_text))
                                                                                                                                                                     ELSE TO_CLOB('')
                                                                                                                                                          END,
                                                                                                                                                          CASE WHEN LENGTH(mel_procedure.maint_proc_text) > 1 
                                                                                                                                                                     THEN TO_CLOB(CONCAT(TO_CLOB('<br /><b>(M)</b><br />'), mel_procedure.maint_proc_text))
                                                                                                                                                                     ELSE TO_CLOB('')
                                                                                                                                                          END
                                                                                                                                                         )
                                                                                                                                       )
             		                                                                                       )
                                                                                                    )
                                                                        ELSE TO_CLOB('')
                                                              END
             	                          END 
                              ELSE TO_CLOB('')
                   END) as c_mel_procedures,
TO_CHAR(CASE WHEN aircraft.ac_typ IN ('B737', 'B777')
                             THEN TO_CLOB(CONCAT(TO_CLOB('<b>MEL Remarks:</b><br />'), mel_detail.mel_remarks))
                             ELSE TO_CLOB('')
                   END) as remarks
FROM (
             SELECT apu_inop.ac_registr as ac_registr,
                           apu_inop.ac_typ as ac_typ,
                           apu_inop.wono as wono,
                           apu_inop.ata as ata,
                           apu_inop.issue_date as issue_date,
                           apu_inop.mel_code as mel_code,
                           apu_inop.wo_due_date as wo_due_date,
                           RTRIM(
                                        XMLAGG(
                                                          XMLELEMENT(E,apu_inop.text, CHR(10) ).EXTRACT('//text()')
                                                          ORDER BY apu_inop.wono
                                                         ).GetClobVal(), CHR(10)
                                       ) as text,
                           apu_inop.hil
             FROM (
                          SELECT wo_header.event_perfno_i as event_perfno_i,
                                        wo_header.ac_registr as ac_registr,
                                        aircraft.ac_typ as ac_typ,
                                        wo_header.workorderno_display as wono,
                                        wo_header.mel_code,
                                        wo_header.hil,
                                        wo_header.ata_chapter as ata,
                                        TO_CHAR(TO_DATE('01.01.1970', 'DD.MM.YYYY') + (wo_header.issue_date+729), 'DD.MM.YYYY') as issue_date,
                                        wo_text_description.header
                                       ||CHR(10)||CASE WHEN wo_text_description.text IS NULL THEN wo_text_description.text_html ELSE wo_text_description.text END
                                       ||CHR(10)||wo_text_action.header||CHR(10)||wo_text_action.text as text,
                                        wo_header_more.mel_chapter as mel_chapter,
                                        wo_header.issue_date as adate,
                                        TO_CHAR(TO_DATE('01.01.1970', 'DD.MM.YYYY') + (wo_transfer.absolute_due_date+729), 'DD.MM.YYYY') as wo_due_date
                          FROM wo_header
                          LEFT JOIN wo_event_link ON wo_header.event_perfno_i=wo_event_link.event_perfno_i
                          LEFT JOIN doc_header ON wo_event_link.event_key_parent=doc_header.docno_i 
                          LEFT JOIN workstep_link ON wo_header.event_perfno_i=workstep_link.event_perfno_i
                          LEFT JOIN wo_text_description ON workstep_link.descno_i=wo_text_description.descno_i
                          LEFT JOIN wo_text_action ON wo_text_action.workstep_linkno_i=workstep_link.workstep_linkno_i
                          LEFT JOIN aircraft ON wo_header.ac_registr=aircraft.ac_registr
                          LEFT JOIN wo_header_more ON wo_header.event_perfno_i = wo_header_more.event_perfno_i
                          LEFT JOIN wo_transfer ON (wo_header.event_perfno_i = wo_transfer.event_perfno_i AND wo_transfer.is_last_transfer = 'Y')
                          WHERE wo_header.state='O'
                                         AND wo_header.type NOT IN ('T', 'PD')
                                         AND aircraft.status<>9
                                         AND (doc_header.doc_type NOT IN ('SB','VSB','EAD') OR doc_header.doc_type IS NULL)
                         ) apu_inop
             WHERE (
                             (
                              (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(inop)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(shutdown)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(maint)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(28).*') AND REGEXP_LIKE(apu_inop.text, '(fuel)','i') AND REGEXP_LIKE(apu_inop.text, '(apu lo pr)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(fuel)','i') AND REGEXP_LIKE(apu_inop.text, '(trim line fault)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(28).*') AND REGEXP_LIKE(apu_inop.text, '(fuel)','i') AND REGEXP_LIKE(apu_inop.text, '(trim line fault)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(bleed)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(36).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(bleed)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(fault)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(24).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(fault)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(shut)','i') AND REGEXP_LIKE(apu_inop.text, '(down)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(24).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(inop)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(24).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(off)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu door)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(неустойч)','i'))
                              OR (REGEXP_LIKE (apu_inop.ata, '^(49).*') AND REGEXP_LIKE(apu_inop.text, '(apu)','i') AND REGEXP_LIKE(apu_inop.text, '(remov)','i'))
                             )
                          OR (
                                  (REGEXP_LIKE(text, '(apu|всу)', 'i'))
                                  AND (
                                             (REGEXP_LIKE (apu_inop.mel_chapter, '^(49).*'))
                                             OR apu_inop.ac_typ='B737' AND (REGEXP_LIKE (apu_inop.mel_chapter, '^(21-02-02|24-02|24-19|26-01|26-06|26-07|26-08-01|26-08-02|26-08-03|26-08-04|26-09-01|26-09-02|28-04).*'))
                                             OR apu_inop.ac_typ='B777' AND (REGEXP_LIKE (apu_inop.mel_chapter, '^(24-31-01|24-21-01|24-21-01-01|24-28-02|24-31-02|26-15-01|26-21-01|26-22-01|26-22-02|28-25-01|28-25-02|28-25-03|36-12-03).*'))
                                             OR apu_inop.ac_typ='A330' AND (REGEXP_LIKE (apu_inop.mel_chapter, '^(24-01-01-01|24-01-01-02|24-01-01-03|24-01-01-04|24-07-01-02|24-23-01|24-32-02|24-38-02|24-38-04|26-01-02-01|26-01-02-02|
                                                                                                 26-01-02-03|26-01-02-31|26-13-01|26-13-02|26-22|28-22|28-27-08|28-29|36-01-03|36-01-04|36-07-01|36-09-02|36-12-01|
                                                                                                 36-12-02|36-12-03|36-22).*'))
                                             OR apu_inop.ac_typ='A32S' AND (REGEXP_LIKE (apu_inop.mel_chapter, '^(24-01-01-03|24-01-01-04|24-23-01|26-01-01-05|26-01-01-06|26-01-01-32|26-01-01-33|26-01-01-34|26-13-01|26-22-01|26-22-02|
                                                                                                 26-22-03|26-22-04|26-22-05|26-22-06|26-22-07|28-07-04-02|28-22-01|28-29-01|36-01-03|36-01-04|36-07-01|36-12-01|36-12-02|
                                                                                                 36-12-03|36-22-03).*'))
                                             OR apu_inop.ac_typ='RRJ' AND (REGEXP_LIKE (apu_inop.mel_chapter, '^(21-21-15|24-23-01|26-12-00|26-12-01|26-12-05|28-22-55|28-25-10|28-40-00-02|36-12-00).*'))
                                            )
                              )
                         )
             GROUP BY apu_inop.ac_typ,
                                  apu_inop.ac_registr,
                                  apu_inop.wono,
                                  apu_inop.ata,
                                  apu_inop.issue_date,
                                  apu_inop.mel_code,
                                  apu_inop.wo_due_date,
                                  apu_inop.mel_code,
                                  apu_inop.hil
            ) i
LEFT JOIN aircraft ON aircraft.ac_registr = i.ac_registr
LEFT JOIN workstep_link ON (workstep_link.event_perfno_i = i.wono AND workstep_link.sequenceno = 1)
LEFT JOIN wo_text_description ON wo_text_description.descno_i = workstep_link.descno_i
LEFT JOIN wo_transfer ON (i.wono = wo_transfer.event_perfno_i AND wo_transfer.is_last_transfer = 'Y')
LEFT JOIN wo_header_more ON i.wono = wo_header_more.event_perfno_i
LEFT JOIN wo_opus ON i.wono = wo_opus.event_perfno_i
LEFT JOIN wo_header_4 on wo_header_4.event_perfno_i = i.wono
LEFT JOIN mel_detail ON mel_detail.mel_detailno_i = wo_header_4.mel_detailno_i
LEFT JOIN mel_tree ON mel_detail.mel_treeno_i = mel_tree.mel_treeno_i
LEFT JOIN mel_procedure ON mel_procedure.mel_treeno_i = mel_tree.mel_treeno_i
LEFT JOIN (
                    SELECT limitation_config.configuration_key as event_perfno_i,
                                  '<b>WO Downgrades (limitations):</b><br />'
                                  ||LISTAGG(TO_CHAR(limitation_type.limitation_type)
                                  ||': '
                                  ||TO_CHAR(limitation_type.description), '; ') WITHIN GROUP (ORDER BY limitation_type.limitation_type, limitation_type.description) as limitations_agg
                    FROM limitation_config
                    LEFT JOIN limitation_type ON limitation_type.limitation_typeno_i = limitation_config.limitation_typeno_i
                    WHERE  limitation_config.mime_type = 'WO'
                    GROUP BY limitation_type.limitation_type, limitation_type.description, limitation_config.configuration_key
                   ) ac_lim ON ac_lim.event_perfno_i = i.wono
WHERE 
                      (
                        mel_tree.version=(
                                                           SELECT MAX (mt.version) 
                                                           FROM mel_tree mt                       
                                                           WHERE mel_tree.mel_treeno_i = mt.mel_treeno_i                             
                                                                          AND mel_tree.revision = mt.revision
                                                          )
                        OR mel_tree.version IS NULL
                       )	
             AND (
                        mel_detail.version=(
                                                              SELECT MAX (md.version)                         
                                                              FROM mel_detail md                         
                                                              WHERE mel_detail.mel_detailno_i = md.mel_detailno_i                               
                                                                             AND mel_detail.revision = md.revision
                                                            )
                        OR mel_detail.version IS NULL
                       )
             AND (
                        mel_procedure.version=(
                                                                       SELECT MAX (mp.version)
                                                                       FROM mel_procedure mp                           	
                                                                       WHERE mel_procedure.mel_treeno_i = mp.mel_treeno_i  
                                                                                      AND mel_procedure.revision = mp.revision
                                                                     )
                        OR mel_procedure.version IS NULL
                       )
