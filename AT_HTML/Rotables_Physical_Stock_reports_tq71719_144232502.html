SELECT part.partno,
       part.description,
       part.vendor,
       project.projectno,
       part.ata_chapter,
       part.mat_class,
       part.mat_type,
       part.repairable,
       part.ac_typ,
       part.special_contract,
       part.tool,
       location.station,
       location.store,
       location.location,
       location.description as loc_desc,
       condition.condition,
       part_fa_entity.average_price,
       rotables.psn,
       rotables.serialno,
       rotables.owner,
       ps.special,
       condition.description cond_desc,
       location.location_type,
              CASE 
            WHEN rotables.mfg_unknown = 'Y' THEN 'UNKNOWN'
            WHEN rotables.mfg_date = '0' THEN 'UNKNOWN'
            ELSE to_char(to_date('01.01.1970', 'dd.mm.yyyy') + (rotables.mfg_date + 729),'dd.mm.yyyy') 
        END as mfg_date, 
       part_fa_entity.purch_price,
       part_fa_entity.average_price,
       part_fa_entity.average_price_func,
       part_fa_entity.fsv,
       part_fa_entity.fsv_func,
       part_fa_entity.fa_qty
FROM part
left JOIN rotables ON part.partno = rotables.partno
left JOIN part_fa_entity ON part.partno = part_fa_entity.partno AND (part_fa_entity.entityno_i = 1)
left JOIN location ON rotables.locationno_i = location.locationno_i
LEFT JOIN project ON rotables.projectno_i = project.projectno_i
LEFT JOIN condition ON rotables.condition = condition.condition
left JOIN (
        select 
                part_special.partno
                , LISTAGG (part_special.special, ';') WITHIN GROUP (ORDER BY part_special.partno) as special
        from part_special
        group by part_special.partno
                
) ps on part.partno = ps.partno
WHERE location.location_type <> -200
AND part.mat_class = 'R'
    AND (rotables.partnonew = ' ' OR  ASCII(rotables.partnonew) = 49824 OR rotables.partnonew IS NULL)
    AND (rotables.ac_registr = ' ' OR rotables.ac_registr = 'TRANSF' OR ASCII(rotables.ac_registr) = 49824)
    AND 1 = 1
