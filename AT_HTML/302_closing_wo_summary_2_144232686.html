SELECT
    wo_header.event_perfno_i,
    wo_header.ac_registr,
    wo_header.state,
    TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (wo_header.issue_date), 'DD.MM.YYYY') AS issue_date,
    TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (wo_header.closing_date), 'DD.MM.YYYY') AS close_date,
    wo_header.ata_chapter as ata_chapter,
    SUBSTR(wo_header.ata_chapter, 1, 2) AS ata_chapter_2,
    wo_header.type,
    CASE WHEN wo_header.HIL IS NULL AND wo_header.mel_code IS NOT NULL THEN 'M:' || wo_header.mel_code
         WHEN wo_header.HIL='Y' THEN ''
        ELSE ''
    END as mel,
    wo_header_more.mel_chapter AS mel_chap,
    wo_header.HIL as DD,
    wo_text_description.header,
    wo_text_description.text,
    aircraft.ac_typ,
    aircraft.ac_model,
    wo_header.closing_date-wo_header.issue_date AS delta_date
FROM wo_header
LEFT JOIN aircraft ON wo_header.ac_registr=aircraft.ac_registr
LEFT JOIN wo_header_more ON wo_header.event_perfno_i=wo_header_more.event_perfno_i
LEFT JOIN workstep_link ON wo_header.event_perfno_i=workstep_link.event_perfno_i and workstep_link.sequenceno=1
LEFT JOIN wo_text_description ON workstep_link.descno_i=wo_text_description.descno_i
WHERE (
    wo_header.closing_date >= (TRUNC(SYSDATE-7) - TO_DATE('31.12.1971', 'DD.MM.YYYY'))
        AND
    wo_header.closing_date<= (TRUNC(SYSDATE) - TO_DATE('31.12.1971', 'DD.MM.YYYY'))
    )
  AND wo_header.state='C'
  AND wo_header.type IN ('P', 'C', 'M')
  AND aircraft.status <> 9
  AND aircraft.ac_registr NOT IN (
      'VP-BAC', 'VP-BET',
      'VP-BOE', 'VQ-BEH',
      'VQ-BEJ', 'VQ-BIT',
      'VQ-BIV'
    )
  AND aircraft.operator = 'AFL'
ORDER BY aircraft.ac_typ
