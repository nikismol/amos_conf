select
    inventory.partno,

from
(select distinct
    rotables.partno as Partno,
    rotables.serialno as Serialno,
    rotables.ac_registr as Ac_Registr,
    rotables.labelno as Labelno,
    TO_CHAR(TO_DATE('01.01.1970', 'dd.Mm.yyyy') + (history.del_date+729), 'dd-Mon-yyyy') as Intall_Date,
    case when rotables.ac_registr=' ' then TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(counter_rh_rem.life_value_msn)/60),'FM999999999900')) || ':'
                            ||TO_CHAR(TRUNC(MOD(TO_NUMBER(counter_rh_rem.life_value_msn),60)),'FM00')
        else TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(counter_rh.life_value_msn + (counter_rh.life_value-counter_rh.life_value_on))/60),'FM999999999900')) || ':'
                            ||TO_CHAR(TRUNC(MOD(TO_NUMBER(counter_rh.life_value_msn + (counter_rh.life_value-counter_rh.life_value_on)),60)),'FM00') end as TSN,
    case when rotables.ac_registr=' ' then TO_CHAR(TO_NUMBER(counter_rc_rem.life_value_csn))
        else TO_CHAR(TO_NUMBER(counter_rc.life_value_csn + (counter_rc.life_value - counter_rc.life_value_on))) end as CSN,
    case when rotables.ac_registr=' ' then ''
        else TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER((counter_rh.life_value-counter_rh.life_value_on))/60),'FM999999999900')) || ':'
                            ||TO_CHAR(TRUNC(MOD(TO_NUMBER((counter_rh.life_value-counter_rh.life_value_on)),60)),'FM00') end as TSI,
    case when rotables.ac_registr=' ' then ''
        else TO_CHAR(TO_NUMBER((counter_rc.life_value - counter_rc.life_value_on))) end as CSI
from rotables
left join location on rotables.locationno_i = location.locationno_i
left join aircraft on rotables.ac_registr = aircraft.ac_registr
left join history on rotables.labelno=history.labelno and rotables.psn=history.psn and history.VM = 'YE'
/*получаю наработки в минутах при установке*/
left join (
    select
        counter_value_m_h.life_value,
        counter_h.life_value_on,
        counter_h.life_value as life_value_msn,
        counter_h.ref_key
    from (
          select counter.ref_key,
                 counter.counterno_i,
                 counter.is_unknown,
                 counter.master_counterno_i,
                 counter_value.on_counter_valueno_i,
                 counter_definition.code,
                 nvl(counter_value.life_value, 0) as life_value,
                 nvl(counter_value_on.life_value, 0) as life_value_on,
                 max (counter_value.readout_date*86400000+counter_value.readout_time) over (partition by counter_value.counterno_i) as max_readout,
                 counter_value.readout_date*86400000+counter_value.readout_time as readout
          from counter
          join counter_template on counter.counter_templateno_i=counter_template.counter_templateno_i
          join counter_definition on counter_template.counter_defno_i=counter_definition.counter_defno_i
          left join counter_value on counter_value.counterno_i=counter.counterno_i
          left join counter_value counter_value_on on counter_value.on_counter_valueno_i=counter_value_on.counter_valueno_i
          where 1=1
                and counter.ref_type='RO'
                and counter_definition.counter_defno_i='1'
                and counter_value.on_counter_valueno_i IS NOT NULL
          order by counter_value.counterno_i
         ) counter_h
    left join counter counter_m_h ON counter_h.master_counterno_i=counter_m_h.counterno_i
    left join counter counter_value_m_h ON counter_m_h.counterno_i=counter_value_m_h.counterno_i
    where 1=1
          and counter_h.readout=counter_h.max_readout
) counter_rh on rotables.psn = counter_rh.ref_key
/*получаю наработки в циклах при установке*/
