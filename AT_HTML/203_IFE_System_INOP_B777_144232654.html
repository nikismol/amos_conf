SELECT  
        TO_CHAR(wo_header.event_perfno_i) AS WO
        ,TO_CHAR(wo_header.ac_registr) AS AC_Registr
        ,TO_CHAR(aircraft.ac_typ) AS AC_Type
        ,CASE  WHEN wo_header.type = 'C' THEN 'Cabin'
                       WHEN wo_header.type = 'P' THEN 'Pirep' 
                       WHEN wo_header.type = 'M' THEN 'Maintenance'
                       ELSE '' END AS WO_Type
        ,CASE WHEN wo_header.mel_code IS NULL 
                     AND wo_header.hil = 'Y'
                     AND wo_opus.free_text IS NOT NULL THEN 'Y' ELSE '' END AS DD
        ,wo_header.issue_sign AS Issue_Sign
        ,to_char(to_date('01.01.1970', 'dd.mm.yyyy') + ( wo_header.issue_date+729),'dd.mm.yyyy') AS Issue_Date 
        ,to_char(to_date('01.01.1970', 'dd.mm.yyyy') + ( wo_header.closing_date+729),'dd.mm.yyyy') AS Closing_Date
        ,wo_header.ata_chapter AS ATA
        ,wo_header.mel_code AS MEL
        ,wo_header.state AS WO_State
        ,part_req.pn AS Req_Parts
        ,dblink.ref AS Refs
        ,wo.step_text
        ,wo.action_text
FROM (

            SELECT  DISTINCT
                    wo_header.event_perfno_i AS event_perfno_i
                    ,LISTAGG (TO_CHAR(TO_DATE('01.01.1970', 'DD.MM.YYYY') + (wo_text_description.created_date+729), 'DD.MM.YYYY')||
                                                                                                    CASE WHEN wo_text_description.text IS NOT NULL 
                                                                                                    THEN ': '||TO_CHAR(wo_text_description.text)||'.' ELSE '' END, CHR(10)||CHR(10))
                    WITHIN GROUP (ORDER BY workstep_link.sequenceno) as step_text
                    ,LISTAGG (TO_CHAR(TO_DATE('01.01.1970', 'DD.MM.YYYY') + (wo_text_action.action_date+729), 'DD.MM.YYYY')||
                                                                                                    CASE WHEN wo_text_action.text IS NOT NULL 
                                                                                                    THEN ': '||TO_CHAR(wo_text_action.text)||'.' ELSE '' END, CHR(10)||CHR(10))
                    WITHIN GROUP (ORDER BY workstep_link.sequenceno) as action_text
            FROM
                    wo_header
            LEFT JOIN workstep_link ON wo_header.event_perfno_i=workstep_link.event_perfno_i
            LEFT JOIN wo_text_description ON workstep_link.descno_i=wo_text_description.descno_i
            LEFT JOIN aircraft ON wo_header.ac_registr=aircraft.ac_registr
            LEFT JOIN wo_header_more ON wo_header.event_perfno_i = wo_header_more.event_perfno_i
            LEFT JOIN wo_text_action ON workstep_link.workstep_linkno_i=wo_text_action.workstep_linkno_i
            WHERE
                   TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + (wo_header.issue_date), 'dd.mm.yyyy') >= SYSDATE-14
                    AND wo_header.type IN ('M', 'C', 'P')
                    AND (
                               wo_header.ata_chapter LIKE ('44%') AND aircraft.ac_typ = 'B777'
          		AND (
					REGEXP_LIKE (wo_text_description.text, '(\sIFE|WIFI|WI-FI|WI FI|MOVIE|MOVIES|MUSIC|VIDEO)','i')
					OR
					REGEXP_LIKE (wo_text_description.text_html, '(\sIFE|WIFI|WI-FI|WI FI|MOVIE|MOVIES|MUSIC|VIDEO)','i')
					OR
					REGEXP_LIKE (wo_text_description.header, '(\sIFE|WIFI|WI-FI|WI FI|MOVIE|MOVIES|MUSIC|VIDEO)','i')
					OR 
					REGEXP_LIKE (wo_text_action.text, '(\sIFE|WIFI|WI-FI|WI FI|MOVIE|MOVIES|MUSIC|VIDEO)','i')
			       )
                        )
	GROUP BY  wo_header.event_perfno_i
    ) wo 
LEFT JOIN wo_header ON wo.event_perfno_i = wo_header.event_perfno_i
LEFT JOIN aircraft ON wo_header.ac_registr = aircraft.ac_registr
LEFT JOIN wo_header_more ON wo_header.event_perfno_i = wo_header_more.event_perfno_i
LEFT JOIN wo_opus ON wo_header.event_perfno_i = wo_opus.event_perfno_i    
LEFT JOIN   (                   
                                SELECT wo_header.event_perfno_i as event_perfno_i,
                                LISTAGG (CASE WHEN LENGTH(part.description) + LENGTH(part_request_2.partno) <= 48
                                                           THEN part.description||' '||part_request_2.partno||' - '||TO_NUMBER(part_request_2.qty)||'ea'
                                                           ELSE SUBSTR(part.description, 1, 42)||'...'||' '||part_request_2.partno||' - '||TO_NUMBER(part_request_2.qty)||'ea'
                                                           END, CHR(10)
                                        ) WITHIN GROUP (ORDER BY part_request_2.part_requestno_i) as pn
                   FROM wo_header
                   JOIN aircraft ON wo_header.ac_registr=aircraft.ac_registr AND aircraft.ac_typ IN ('A32S', 'A330', 'B737','B777')
                   JOIN workstep_link ON wo_header.event_perfno_i=workstep_link.event_perfno_i and wo_header.state='O' AND wo_header.type NOT IN ('S', 'T', 'PD')
                   JOIN part_request_2 ON workstep_link.workstep_linkno_i=part_request_2.event_key
                   LEFT JOIN part ON part_request_2.partno=part.partno
                   WHERE part_request_2.event_type = 'WSL'
                   GROUP BY wo_header.event_perfno_i
            ) part_req ON wo_header.event_perfno_i=part_req.event_perfno_i
LEFT JOIN   ( 
                SELECT
                dbl.event_perfno_i
                ,LISTAGG (dbl.ref_type||' '||dbl.description, CHR(10)) WITHIN GROUP (ORDER BY dbl.ref_type) as ref  
                FROM (
                        SELECT
                                workstep_link.event_perfno_i,
                                db_link.description,
                                db_link.ref_type
                        FROM workstep_link
                        JOIN db_link ON TO_CHAR(workstep_link.descno_i) = TO_CHAR(db_link.source_pk) AND db_link.source_type='WS' 
                        
                        UNION

                        SELECT
                                wo_header.event_perfno_i,
                                db_link.description,
                                db_link.ref_type
                        FROM wo_header
                        JOIN db_link ON TO_CHAR(wo_header.event_perfno_i) = TO_CHAR(db_link.source_pk) AND db_link.source_type='WO' 
                    ) dbl
                GROUP BY dbl.event_perfno_i
            ) dblink ON wo_header.event_perfno_i = dblink.event_perfno_i
