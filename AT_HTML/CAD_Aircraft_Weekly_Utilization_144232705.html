SELECT aircraft.ac_model,
              TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(SUM(ac_utilization.daily_hours))/60),'FM999999999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(SUM(ac_utilization.daily_hours)),60)),'FM00') as weekly_hours,
              TO_NUMBER(SUM(ac_utilization.daily_cycles)) as weekly_cycles
FROM ac_utilization
JOIN aircraft ON ac_utilization.ac_registr=aircraft.ac_registr
WHERE 1=1
              AND ac_utilization.null_leg_commited = 0
              AND ac_utilization.recno > 0
              AND TO_DATE('31.12.1971','DD.MM.YYYY') + ac_utilization.validity BETWEEN TRUNC(TRUNC(SYSDATE, 'IW' ) - 1, 'IW') /* start_last_week */
                                                                                                                                                                      AND  TRUNC(SYSDATE, 'IW') - 1 /* end_last_week */
GROUP BY aircraft.ac_model


UNION

SELECT 'TOTAL' as ac_model,
              TO_CHAR(TO_CHAR(TRUNC(TO_NUMBER(SUM(ac_utilization.daily_hours))/60),'FM999999999900')) || ':' ||TO_CHAR(TRUNC(MOD(TO_NUMBER(SUM(ac_utilization.daily_hours)),60)),'FM00') as weekly_hours,
              TO_NUMBER(SUM(ac_utilization.daily_cycles)) as weekly_cycles
FROM ac_utilization
JOIN aircraft ON ac_utilization.ac_registr=aircraft.ac_registr
WHERE 1=1
              AND ac_utilization.null_leg_commited = 0
              AND ac_utilization.recno > 0
              AND TO_DATE('31.12.1971','DD.MM.YYYY') + ac_utilization.validity BETWEEN TRUNC(TRUNC(SYSDATE, 'IW' ) - 1, 'IW') /* start_last_week */
                                                                                                                                                                      AND  TRUNC(SYSDATE, 'IW') - 1 /* end_last_week */
ORDER BY 1
