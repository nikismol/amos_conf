SELECT
    part_forecast.event_display AS event
    ,aircraft.ac_typ
    ,part_forecast.receiver AS receiver
    ,CASE WHEN REGEXP_LIKE(part_forecast.partno, '^[0-9]+$') THEN chr(39)||part_forecast.partno ELSE part_forecast.partno END AS partno
    ,part_forecast.part_description as part_description
    ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + part_forecast.expected_date, 'DD.MM.YYYY') AS due_date
    ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + part_forecast.planned_date, 'DD.MM.YYYY') AS planned_date
    ,TO_NUMBER(part_forecast.requested_qty) AS requested_qty
    ,part_forecast.measure_unit AS measure_unit
    , CASE
        WHEN part_forecast.planned_date IS not NULL THEN TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (part_forecast.planned_date-4), 'DD.MM.YYYY')
        END AS delivery_date
    ,od_header.orderno
    ,od_header.ext_state
    ,TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + od_header.orderdate, 'DD.MM.YYYY') AS orderdate
FROM part_forecast
LEFT JOIN part_request_2 ON part_forecast.part_requestno_i=part_request_2.part_requestno_i and part_request_2.event_type='WSL'
LEFT JOIN allocation_link ON part_request_2.part_requestno_i=allocation_link.source_key AND allocation_link.ref_type='ORDER_DE'
LEFT JOIN aircraft ON part_forecast.receiver=aircraft.ac_registr
LEFT JOIN od_detail ON allocation_link.ref_key=od_detail.detailno_i
LEFT JOIN od_header ON od_detail.orderno_i=od_header.orderno_i AND od_header.order_type='P' AND od_header.ext_state IN ('O', 'D')
WHERE
   (
       part_forecast.part_request_part_cond <> 'US' OR part_forecast.part_request_part_cond IS NULL
       )
  AND aircraft.status=0
  AND
    (
        part_forecast.EXPECTED_DATE >= FLOOR(TRUNC(sysdate, 'dd')-to_date('19711231','yyyymmdd'))
        AND part_forecast.EXPECTED_DATE <= FLOOR(TRUNC(ADD_MONTHS(sysdate, 12), 'dd')-to_date('19711231','yyyymmdd'))
        )
  AND
    (
        (/*A32S*/
            part_forecast.event_display LIKE ('A20-25-0622-01%')
                OR part_forecast.event_display LIKE ('A20-25-0622-02%')
                OR part_forecast.event_display LIKE ('A20-25-0622-03%')
            )
        OR
        (/*A330*/
            part_forecast.event_display LIKE ('A30-25-0079-01%')
                OR part_forecast.event_display LIKE ('A30-25-0079-02%')
                OR part_forecast.event_display LIKE ('A30-25-0079-03%')
                OR part_forecast.event_display LIKE ('A30-25-0079-04%')
            )
        OR
        (/*A350*/
            part_forecast.event_display LIKE ('252861-00M01-AFL%')
                OR part_forecast.event_display LIKE ('252861-00M02-AFL%')
                OR part_forecast.event_display LIKE ('252861-00M03-AFL%')
                OR part_forecast.event_display LIKE ('252861-00M04-AFL%')
            )
        OR
        (/*B737*/
            part_forecast.event_display LIKE ('B3-25-0043-01%')
                OR part_forecast.event_display LIKE ('B3-25-0043-02%')
                OR part_forecast.event_display LIKE ('B3-25-0043-03%')
            )
        OR
        (/*B777*/
            part_forecast.event_display LIKE('%B7-25-0066/01%')
                OR part_forecast.event_display LIKE ('B7-25-0066/02%')
                OR part_forecast.event_display LIKE ('B7-25-0066/04%')
                OR part_forecast.event_display LIKE ('B7-25-0066/05%')
            )
        )
ORDER BY aircraft.ac_typ, part_forecast.receiver, part_forecast.partno, part_forecast.event_display
