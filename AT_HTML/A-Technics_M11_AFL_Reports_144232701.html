SELECT
    TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (h.del_date), 'DD.MM.YYYY') as utc_date,
    p.description part_description,
    h.partno partnumber,
    h.serialno serialnumber,
    h.batchno batchnumber,
    p.measure_unit,
    h.qty quantity,
    h.mat_class,
   CASE WHEN h.condition = 'S' THEN 'SERVICABLE / ИСПРАВНО'
        WHEN h.condition = 'US' THEN 'UNSERVICABLE / НЕИСПРАВНО'
        WHEN h.condition = 'OH' THEN 'OVERHAULED / КАПИТАЛЬНЫЙ РЕМОНТ'
        WHEN h.condition = 'N' THEN 'NEW'
        WHEN h.condition = 'R' THEN 'REPAIRED / ТЕКУЩИЙ РЕМОНТ'
        WHEN h.condition = 'IT' THEN 'INSPECTED / TEST/ ПРОВЕРЕНО / ИСПЫ'
        ELSE h.condition
   END conditions,
   CASE WHEN h.vm = 'AA' THEN 'STORE TO A/C'
        WHEN h.vm = 'YE' THEN 'INSTALLATION'
        WHEN h.vm = 'BN' THEN 'BATCHNO CHANGE'
        WHEN h.vm = 'CS' THEN 'CU.LOAN START'
        WHEN h.vm = 'CR' THEN 'CANCEL RECEIVING'
        WHEN h.vm = 'I' THEN 'INVENT.CORR'
        WHEN h.vm = 'AS' THEN 'STORE TO SHOP'
        ELSE h.vm
   END operation_type,
    h.voucherno,
    h.ac_registr,
   CASE WHEN h.vm = 'CS' THEN adr.name
   END receivers
FROM history h
LEFT JOIN part p ON h.partno = p.partno
LEFT JOIN address adr ON h.receiver = adr.vendor
WHERE h.vm IN (
         'AA',
         'YE',
         'BN',
         'CS',
         'CR',
         'I',
         'AS'
      )
  AND
    (h.del_date between FLOOR(trunc(ADD_MONTHS(SYSDATE, -1), 'MM')-to_date('19711231','yyyymmdd'))
        and FLOOR(trunc(SYSDATE, 'MM') -1-to_date('19711231','yyyymmdd'))
        )
  AND h.station||h.store IN (
         'SVOHANGAR_4',
         'SVOHANGAR_5',
         'SVOAT_ACC',
         'SVOAT_BASE',
         'SVOAT_CABIN',
         'SVOAT_FLUID',
         'SVOAT_WHEEL',
         'SHRM5010'
      )
  AND h.qty < 0
