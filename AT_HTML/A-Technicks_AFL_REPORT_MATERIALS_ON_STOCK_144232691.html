select
    t.bar_code,
    t.partno,
    t.Serial_or_Batch,
    TO_CHAR(TO_DATE('31.12.1971', 'DD.MM.YYYY') + (t.Expire_Date), 'DD.MM.YYYY') Expire_Date,
    part.ac_typ as FA_Type,
    part.ata_chapter,
    part.repairable,
    part.description,
    part.mat_class,
    part.mat_type,
    t.qty,
    part.measure_unit,
    t.station,
    t.store,
    t.location,
    t.owner
from (
    select
        r.rec_detailno_i as bar_code,
        r.partno, r.serialno as Serial_or_Batch,
        r.expiry_date as Expire_Date,
        1 qty,
        l.station,
        l.store,
        l.location,
        r.owner
    from rotables r, location l
    where r.locationno_i = l.locationno_i
    union all
    select
        c.recdetailno_i as bar_code,
        c.partno, c.batchno as Serial_or_Batch,
        c.expire_date as Expire_Date,
        c.qty, l.station,
        l.store,
        l.location,
        c.owner
    from consumables c, location l
    where c.locationno_i = l.locationno_i ) t
JOIN part ON t.partno = part.partno
where t.location <>'OUTGOING'
  and t.station||t.store IN (
                             'SVOHANGAR_4',
                             'SVOHANGAR_5',
                             'SVOAT_ACC',
                             'SVOAT_BASE',
                             'SVOAT_CABIN',
                             'SVOAT_FLUID',
                             'SVOAT_WHEEL',
                             'SHRM5010'
                            )
ORDER BY t.owner
