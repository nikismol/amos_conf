SELECT 
       part_forecast.partno,
       part_forecast.part_description,
       part_forecast.mat_class,
       part_forecast.part_ata_chapter,
       part_forecast.requested_qty,
       (select amosserver_prod.get_stock_incl_alt(part_forecast.partno, 'ALL_STATIONS', 'AFL_ONLY_STOCK') from dual) as stock_incl_alt,
       part_forecast.measure_unit,
       part_forecast.receiver as ac_registr,
       aircraft.ac_typ as ac_typ,
       part_forecast.event_perfno_i as wo,
       part_forecast.event_display as event_description,
       TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + part_forecast.valid_due_date, 'dd.mm.yyyy') as valid_due_date,
       TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') + part_forecast.part_request_created_date, 'dd.mm.yyyy') as part_request_created_date,
       TO_CHAR(TO_DATE('31.12.1971', 'dd.mm.yyyy') +  part_forecast.expected_date, 'dd.mm.yyyy') as expected_date,
       part_forecast.wpno,
       od.orderno
FROM part_forecast
JOIN wo_header ON part_forecast.display_event_key = wo_header.event_perfno_i
JOIN wp_header ON part_forecast.wpno_i = wp_header.wpno_i 
LEFT JOIN aircraft ON part_forecast.receiver = aircraft.ac_registr
LEFT JOIN (SELECT od_detail.partno,
                                 LISTAGG(od_header.orderno, ', ') WITHIN GROUP (ORDER BY od_header.orderno) as orderno
                   FROM od_detail
                   JOIN od_header ON od_detail.orderno_i = od_header.orderno_i AND od_header.ext_state IN ('O', 'PR')
                   WHERE 1=1
                                 AND od_detail.ext_state IN ('O', 'PR')
                                 AND od_header.order_type = 'P'
                   GROUP BY od_detail.partno
                  ) od ON part_forecast.partno = od.partno
WHERE 1=1
              AND part_forecast.display_event_type = 'WO'
              AND part_forecast.receiver_type = 'AC'
              AND wo_header.event_type = 'W'
              AND wp_header.maintprov_address = '0011026298'
              AND SUBSTR(part_forecast.part_ata_chapter, 1, 2) <> '25'
