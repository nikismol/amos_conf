Получение historyno_i для выгрузки документов
SELECT
    history.partno
    ,history.serialno
    ,history.batchno
    ,history.historyno_i
    ,history.vm
FROM history
WHERE history.voucherno = '@VAR.voucherno@'


Получение recdetailno_i для прописывания мэппинга
SELECT 
    history.partno
    ,history.batchno
    ,history.recdetailno_i
FROM history
WHERE history.voucherno = '@VAR.voucherno@'
SQL запрос для переноса локаций
SELECT
    location.station AS STATION,
    location.store AS STORE,
    location.location AS LOCATION,
    CASE WHEN location.location_restriction=0 THEN 'S'
        WHEN location.location_restriction=1 THEN 'R'
        ELSE 'I' END AS RESTRICTION,
    location.description,
    location.location_type AS LOC_TYPE,
    location.location_storage_type AS LOCATION_STORAGE_TYPE,
    location.owner AS OWNER,
    location.max_pn_qty AS MAX_PN_QTY,
    '' as BASKET_PROP,
    '' as DROPOFF_PROP,
    location.text AS TEXT
FROM location
WHERE location.station = '@VAR.STATION@'
    AND location.store = '@VAR.STORE@'
SQL запрос для переноса парт линков 
SELECT
    part_location.partno AS PARTNO,
    location.station AS STATION,
    location.store AS STORE,
    location.location AS LOCATION,
    part_location.is_main AS IS_MAIN,
    CASE WHEN location.location_restriction=0 THEN 'S'
         WHEN location.location_restriction=1 THEN 'R'
         ELSE 'I' END AS RESTRICTION,
    location.description AS DESCRIPTION,
    part_location.max_qty AS MAX_QTY
FROM part_location
LEFT JOIN location ON part_location.locationno_i=location.locationno_i
WHERE location.station = '@VAR.STATION@'
 AND location.store = '@VAR.STORE@'
