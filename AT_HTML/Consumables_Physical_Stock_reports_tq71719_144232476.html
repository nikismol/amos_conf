SELECT part.partno,
       part.description,
       part.vendor,
       project.projectno,
       part.ata_chapter,
       part.mat_class,
       part.mat_type,
       part.repairable,
       part.ac_typ,
       part.special_contract,
       part.tool,
       location.station,
       location.store,
       location.location,
       location.description,
       condition.condition,
       part.fixed_asset,
       part_fa_entity.average_price,
       part_fa_entity.fsv,
       consumables.batchno,
       consumables.owner,
       condition.description,
       consumables.recdetailno_i,
       ps.special,
       consumables.qty,
       part_fa_entity.purch_price,
       part_fa_entity.average_price,
       part_fa_entity.average_price_func,
       part_fa_entity.fsv,
       part_fa_entity.fsv_func,
       part_fa_entity.fa_qty,
       location.location
FROM part
left JOIN (
        select 
                part_special.partno
                , LISTAGG (part_special.special, ';') WITHIN GROUP (ORDER BY part_special.partno) as special
        from part_special
        group by part_special.partno          
) ps on part.partno = ps.partno
JOIN consumables ON part.partno = consumables.partno
JOIN part_fa_entity ON part.partno = part_fa_entity.partno AND (part_fa_entity.entityno_i = 1)
JOIN location ON consumables.locationno_i = location.locationno_i
LEFT JOIN project ON consumables.projectno_i = project.projectno_i
LEFT JOIN condition ON consumables.condition = condition.condition
WHERE location.location_type <> -200
      AND consumables.qty > 0
      AND 1 = 1
      AND part.mat_class <> 'R'
