Настройка селектора в APN 1404

  • В Configuration Detail выбираем подвкладку selector и в окне Export Selector заполнить данные:

  • Ниже представлен SQL запрос для указания в Selector Options:

    select 
    wo.workorderno
from 
    (
        select 
            woh.event_perfno_i workorderno,
            woh.ata_chapter,
            woh.state,
            woh.ac_registr,
            to_date('31.12.1971', 'dd.mm.yyyy') + woh.issue_date issue_date,
            to_date('31.12.1971', 'dd.mm.yyyy') + woh.closing_date closing_date,
            woh.issue_station,
            woh.type,
            woh.mel_code,
            to_date('31.12.1971', 'dd.mm.yyyy') + woh.mutation + woh.mutation_time / 24 / 60 / 60 / 1000 last_mutation
        from amosserver_prod.wo_header woh
        where woh.type in ('P', 'C', 'M', 'S')
          and woh.mel_code is not null
          and mutator not in ('MIGRUSR', 'RAMAX_AC', 'ADMINRMX', 'TAH_UPDD')
          and mutation > 16586
          and ac_registr != 'COMP') wo
        left join (
            select 
                success last_transfer_success,
                error_message,
                source_key_i,
                to_date('31.12.1971', 'dd.mm.yyyy') + trf_date + trf_time / 24 / 60 last_trf_date_time,
                cnt_per_source_key
            from (
                select 
                    h.*,
                    max(histno_i) over(partition by source_key_i) max_histno_i,
                    count(1) over(partition by source_key_i) cnt_per_source_key
                from amosserver_prod.aim_transfer_history h
                where service = 'TRANSFER_WORKORDER'
                  and source_type = 'WO'
                  and NVL(aim_configno_i,100500) IN (608, 412, 100500)
                ) t
            where t.histno_i = max_histno_i) t
            on wo.workorderno = t.source_key_i
where (trunc(last_mutation, 'mi') - nvl(last_trf_date_time, to_date('31.12.1971', 'dd.mm.yyyy'))) * 24 * 60 > 1
   or nvl(last_transfer_success, 'N') != 'Y'
