SELECT 
  wp_header.wpno,
  wo_header.ac_registr,
  wp_sequence.seqno_i,
  wo_header.event_perfno_i as met_link,
  pickslip_booked.pickslipno, 
  pickslip_booked.partno,
  pickslip_booked.serialno,
  pickslip_booked.batchno,
  CASE WHEN dbl.linkno_i IS NULL THEN 'Certificate_not_found' 
    ELSE
      TO_CHAR('\\msk.aeroflot.ru\amos\PROD\WEB_DRIVE\')||
      CASE WHEN (dbl.link_type = 'D') THEN REPLACE(dbl.destination_key, '/', '\') ELSE TO_CHAR(ste_content_list.engine_id) END
    END as certificate_path
FROM wo_header
LEFT JOIN wp_assignment ON wp_assignment.event_perfno_i = wo_header.event_perfno_i
LEFT JOIN wp_sequence ON wo_header.event_perfno_i = wp_sequence.event_perfno_i AND wp_sequence.status = 0
LEFT JOIN wp_header ON wp_header.wpno_i = wp_assignment.wpno_i
LEFT JOIN workstep_link ON workstep_link.event_perfno_i = wo_header.event_perfno_i
LEFT JOIN wo_text_action ON wo_text_action.workstep_linkno_i = workstep_link.workstep_linkno_i
LEFT JOIN part_request_2 ON part_request_2.event_type = 'WSL' AND part_request_2.event_key = workstep_link.workstep_linkno_i
LEFT JOIN pickslip_detail ON (pickslip_detail.part_requestno_i = part_request_2.part_requestno_i)
LEFT JOIN pickslip_booked ON pickslip_booked.recno = pickslip_detail.recno AND pickslip_booked.pickslipno = pickslip_detail.pickslipno
LEFT JOIN history ON history.recdetailno_i = pickslip_booked.recdetailno_i AND history.vm = 'B1'
LEFT JOIN rotables ON rotables.psn = history.psn
LEFT JOIN certificate ON certificate.source_type = 'RO' AND certificate.source_key = rotables.psn
LEFT JOIN db_link dbl_cert ON dbl_cert.source_type = 'CERT' AND dbl_cert.source_pk = certificate.certificate_i
LEFT JOIN db_link dbl_rec ON dbl_rec.source_type = 'RD' AND dbl_rec.source_pk = pickslip_booked.recdetailno_i
LEFT JOIN db_link dbl ON (dbl_rec.linkno_i IS NULL AND dbl_cert.linkno_i = dbl.linkno_i) OR (dbl_rec.linkno_i IS NOT NULL AND dbl_rec.linkno_i = dbl.linkno_i)
LEFT JOIN ste_content_list ON dbl.link_type = 'S' AND TO_CHAR(ste_content_list.storage_id) = dbl.destination_key
WHERE
workstep_link.workstep_state = 'C'
AND wo_text_action.mutation = FLOOR(sysdate-to_date('19711231','yyyymmdd'))-1
AND pickslip_booked.pickslipno IS NOT NULL
AND pickslip_booked.qty_booked > 0
