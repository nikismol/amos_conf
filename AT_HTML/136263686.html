Настройка селектора в APN 1404

  • В Configuration Detail выбираем подвкладку selector и в окне Export Selector заполнить данные:

  • Ниже представлен SQL запрос для указания в Selector Options:

    SELECT 
    concat('RD.', od_rec_detail.recdetailno_i)
FROM od_rec_detail
JOIN od_rec_header ON od_rec_header.receivingno = od_rec_detail.receivingno
JOIN od_header ON od_rec_detail.orderno = od_header.orderno
JOIN history ON od_rec_detail.recdetailno_i = history.recdetailno_i
LEFT JOIN aim_transfer_history ON od_rec_detail.recdetailno_i = aim_transfer_history.source_key_i /* Transfer history */
    AND aim_transfer_history.success = 'Y' AND aim_transfer_history.source_type = 'RD'
    AND aim_transfer_history.service = 'TRANSFER_RECEIVING'
    AND aim_transfer_history.aim_configno_i IN ('7')
    AND aim_transfer_history.source_key_i IS NOT NULL
WHERE 1=1
  AND aim_transfer_history.source_key_i IS NULL  /* не выгружались ранее */
  AND to_date('31.12.1971', 'dd.mm.yyyy') + od_rec_detail.created_date >= TRUNC(sysdate - 30)
  AND (
      od_header.order_type IN ('P', 'PX', 'PR', 'PU', 'L', 'LU', 'Y', 'YU', 'RC', 'X', 'CP')
          OR
      od_header.order_type IN ('R') AND od_header.vendor != 'CLOSED'
      ) 
  AND od_rec_header.awbno NOT IN ('RELEASE REPAIRABLE', 'RELEASE ROTABLE')
  AND od_rec_detail.recdetailno_i > 0
  AND od_rec_detail.status <>  9 /* не сторно */
  AND NVL(history.higher_partno, CHR(160)) in (CHR(49824), CHR(160)) /*history.higher_partno is null */
  AND od_rec_detail.recdetailno_i not in (583145, 583143, 583144, 915419, 915420)
GROUP BY od_rec_detail.recdetailno_i

Настройка селектора в APN 1404

  • В Configuration Detail выбираем подвкладку selector и в окне Export Selector заполнить данные:

  • Ниже представлен SQL запрос для указания в Selector Options:

    select 
    concat('RD.', dt.recdetailno_i)
from (
    select 
        created_date, 
        recdetailno_i, 
        receivingno, 
        orderno, 
        status 
    from od_rec_detail
    ) dt
    join (
        select 
            awbno, 
            receivingno 
        from od_rec_header
        ) hr on hr.receivingno = dt.receivingno
    left join /* Transfer history */ (
        select 
            hist.source_key_i
        from aim_transfer_history hist
        where hist.success = 'Y'
          and hist.source_type = 'RD'
          and hist.service = 'TRANSFER_RECEIVING'
          and hist.aim_configno_i = '810'
          and hist.source_key_i is not null
        group by hist.source_key_i
        ) hist on dt.recdetailno_i = hist.source_key_i
    join od_header ord on dt.orderno = ord.orderno
    join history hist2 on dt.recdetailno_i = hist2.recdetailno_i
where
    hist.source_key_i is null 
  and  
    /* не выгружались ранее */ 
    dt.created_date > to_date('14.06.2019', 'dd.mm.yyyy') - to_date('31.12.1971', 'dd.mm.yyyy') 
  and /* созданы после 29.05.2017 */
    ord.order_type in ('P', 'PX', 'PR', 'PU', 'L', 'LU', 'Y', 'YU', 'R',  'X') 
  and /* тип заказа */ hr.awbno != 'RELEASE REPAIRABLE' 
  and hr.awbno != 'RELEASE ROTABLE' 
  and /* не релиз репирабле */ dt.recdetailno_i > 0 
  and /* не пустое */
    /*	dt.created_date = trunc(sysdate)-to_date('31.12.1971', 'dd.mm.yyyy') and   Создана сегодня */
    dt.status !=  9 
  and /* не сторно */ /*hist2.higher_partno is null */
    nvl(hist2.higher_partno, chr(160)) in (chr(49824), chr(160))
  AND dt.recdetailno_i != 62515 /* buggy rec from aug17 */
group by dt.recdetailno_i
