SELECT 
    wo_event_link.event_perfno_i
FROM wo_event_link
JOIN wo_header ON wo_event_link.event_perfno_i=wo_header.event_perfno_i
WHERE wo_event_link.event_key_parent IN (
    SELECT docno_i
    FROM doc_header
	WHERE docno=:name
    )
AND wo_event_link.pending_status IN (
    SELECT MIN(wel.pending_status)
        FROM wo_event_link wel
        JOIN wo_header woh ON wo_event_link.event_perfno_i = woh.event_perfno_i
        WHERE wel.event_key_parent = wo_event_link.event_key_parent 
          AND wel.event_type = 'TCEFFLI'
          AND woh.ac_registr = wo_header.ac_registr
        GROUP BY woh.ac_registr)

event_perfno_inumber(12,0)WOForeign key to current pending event.
SELECT
    wo_event_link.event_perfno_i
FROM wo_event_link
JOIN wo_header ON wo_event_link.event_perfno_i=wo_header.event_perfno_i
JOIN mevt_header ON mevt_header.mevt_headerno_i = wo_event_link.mevt_headerno_i AND mevt_header.mevt_type = 'DO'
WHERE wo_event_link.event_key_root IN (
    SELECT d_docno_i
    FROM doc_header
    LEFT JOIN (
    SELECT
        docno_i,
        d_docno_i
    FROM (
        SELECT DISTINCT
            parrent.docno_i,
            sub.docno,
            sub.docno_i AS d_docno_i,
            sub.revision,
            sub.doc_type
        FROM (
            SELECT
                LEVEL AS lev,
                id AS docno_i,
                pid,
                item.docno,
                item.revision,
                CONNECT_BY_ISLEAF AS IsLea,
                CONNECT_BY_ROOT item.docno_i AS root,
                item.write_off_logic,
                doc_type
            FROM (
                SELECT
                    main.id,
                    CASE
                        WHEN main.id = main.parrent_id THEN NULL
                        ELSE main.parrent_id
                    END AS pid,
                    header.docno_i,
                    header.docno,
                    revision,
                    write_off.write_off_logic,
                    header.doc_type
                FROM (
                    SELECT DISTINCT
                        tree.docno_i AS id,
                        CASE
                            WHEN (
                                SELECT COUNT(doc_signoff_tree.ref_docno_i) AS cnt
                                FROM doc_signoff_tree
                                WHERE tree.docno_i = doc_signoff_tree.ref_docno_i
                            ) > 0 THEN tree.docno_i
                            ELSE NULL
                        END AS parrent_id,
                        docno
                    FROM
                        doc_signoff_tree tree
                    JOIN
                        doc_header doc ON tree.docno_i = doc.docno_i

                    UNION

                    SELECT
                        tree.ref_docno_i AS id,
                        tree.docno_i AS parrent_id,
                        doc.docno
                    FROM
                        doc_signoff_tree tree
                    JOIN
                        doc_header doc ON tree.ref_docno_i = doc.docno_i
                ) main
                JOIN
                    doc_header header ON main.id = header.docno_i
                /* Document write-off logic */
                LEFT JOIN (
                    SELECT
                        CASE
                            WHEN COUNT(docno_i) > '0' THEN 'Y'
                            ELSE 'N'
                        END AS write_off_logic,
                        docno_i
                    FROM
                        doc_signoff_tree
                    WHERE
                        docno_i = ref_docno_i
                    GROUP BY
                        docno_i
                ) write_off ON header.docno_i = write_off.docno_i
            ) item
            CONNECT BY PRIOR item.id = item.pid
            ORDER SIBLINGS BY item.id, revision
        ) sub
        JOIN
            doc_header parrent ON sub.root = parrent.docno_i
        WHERE
            pid IS NOT NULL
            AND islea = 1
    ) doc
) sign_off_doc ON doc_header.docno_i = sign_off_doc.docno_i
    WHERE doc_header.docno_i=:name
    )
AND wo_header.ac_registr=:ac_reg
AND wo_event_link.PENDING_STATUS >= 0
AND wo_event_link.planable_status=0
AND rownum=1
ORDER BY wo_event_link.created_date, wo_event_link.mutation_time
